## Enunciado

Reescribe el validador del Proyecto 1 con Pandas, **sin bucles por registro**, en `ejercicios/m13-e7/validador_pandas.py`:

1. `leer(carpeta)` → `(cdr, clientes)` con los identificadores como texto y sin convertir los vacíos en faltantes (`keep_default_na=False`).
2. `clasificar(cdr, clientes)` → Series con el primer error de cada registro o `"valido"`.
3. `resumen(cdr, clasificacion)` → las mismas claves que la sección `"cdr"` de `esperado.json` (incluidos minutos, megas, SMS y líneas con consumo).

## Solución

```python validador_pandas.py
"""Ejercicio m13-e7: validador completo con Pandas que escribe sus reportes.

Uso:
    python ejercicios/m13-e7/validador_pandas.py datos/10MB reportes/pandas
"""

from __future__ import annotations

import json
import sys
from pathlib import Path

import numpy as np
import pandas as pd

PRIORIDAD = ["duplicado", "linea_vacia", "linea_desconocida", "tipo_desconocido", "duracion_negativa", "fecha_invalida"]
ORDEN_ESPERADO = ["duplicado", "duracion_negativa", "fecha_invalida", "linea_vacia", "linea_desconocida", "tipo_desconocido"]


def leer(carpeta: Path) -> tuple[pd.DataFrame, pd.DataFrame]:
    cdr = pd.read_csv(
        carpeta / "cdr.csv", sep="|", keep_default_na=False,
        dtype={"linea_origen": "string", "linea_destino": "string", "tipo": "string", "destino": "string"},
    )
    clientes = pd.read_csv(carpeta / "clientes.csv", encoding="latin-1", dtype={"linea": "string"})
    return cdr, clientes


def clasificar(cdr: pd.DataFrame, clientes: pd.DataFrame) -> pd.Series:
    fecha = pd.to_datetime(cdr["fecha_inicio"], format="%Y-%m-%d %H:%M:%S", errors="coerce")
    condiciones = [
        cdr.duplicated("cdr_id"),
        cdr["linea_origen"] == "",
        ~cdr["linea_origen"].isin(clientes["linea"]),
        ~cdr["tipo"].isin(["VOZ", "SMS", "DATOS"]),
        cdr["duracion_seg"] < 0,
        fecha.isna(),
    ]
    return pd.Series(np.select(condiciones, PRIORIDAD, default="valido"), index=cdr.index)


def resumen(cdr: pd.DataFrame, clasificacion: pd.Series) -> dict:
    conteo = clasificacion.value_counts()
    validos = cdr[clasificacion == "valido"]
    voz = validos[validos["tipo"] == "VOZ"]
    minutos = (voz["duracion_seg"] + 59) // 60
    internacional = voz["destino"] == "INTERNACIONAL"
    por_tipo = validos["tipo"].value_counts()
    return {
        "registros_totales": int(len(cdr)),
        "registros_validos": int(len(validos)),
        "errores": {e: int(conteo.get(e, 0)) for e in ORDEN_ESPERADO},
        "validos_por_tipo": {t: int(por_tipo.get(t, 0)) for t in ["VOZ", "SMS", "DATOS"]},
        "minutos_nacionales": int(minutos[~internacional].sum()),
        "minutos_internacionales": int(minutos[internacional].sum()),
        "mb_totales": round(float(validos.loc[validos["tipo"] == "DATOS", "mb"].sum()), 3),
        "sms_totales": int((validos["tipo"] == "SMS").sum()),
        "lineas_con_consumo": int(validos["linea_origen"].nunique()),
    }


def validar(carpeta: Path, salida: Path) -> dict:
    salida.mkdir(parents=True, exist_ok=True)
    cdr, clientes = leer(carpeta)
    clasificacion = clasificar(cdr, clientes)
    invalidos = clasificacion != "valido"
    cdr[invalidos].assign(motivo=clasificacion[invalidos]).to_csv(salida / "rechazados.csv", sep="|", index=False)
    datos = resumen(cdr, clasificacion)
    (salida / "resumen.json").write_text(json.dumps(datos, indent=2, ensure_ascii=False), encoding="utf-8")
    return datos


if __name__ == "__main__":
    print(validar(Path(sys.argv[1]), Path(sys.argv[2])))
```

## Explicación

El validador completo cabe en unas 60 líneas, y cada regla de negocio es una línea legible. Pero el cambio más grande es de rendimiento y de modelo mental:

| | Proyecto 1 (Python puro) | Pandas |
|---|---|---|
| Modelo | Un registro a la vez | Columnas completas |
| Memoria con 1 GB | Pocos MB (+ IDs vistos) | Varios GB: todo el archivo en RAM |
| Velocidad | ~1 millón de registros/s | Varias veces más rápido |
| Con 20 GB | Funciona (lento) | No cabe en 16 GB |

Ninguno es "el mejor": Python puro escala en memoria, Pandas en velocidad y claridad. En el módulo 14 combinarás ambos (Pandas por bloques), y en el 16 verás herramientas que tienen lo mejor de los dos.
