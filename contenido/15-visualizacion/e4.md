def ref(s):
    q1, q3 = np.percentile(s, [25, 75])
    iqr = q3 - q1
    return q1, q3, iqr, ((s < q1 - 1.5 * iqr) | (s > q3 + 1.5 * iqr)).sum()

## Solución

```python
import numpy as np
import pandas as pd

cdr = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"}, keep_default_na=False)


def describir(serie):
    q1, q3 = serie.quantile([0.25, 0.75])
    iqr = q3 - q1
    inferior, superior = q1 - 1.5 * iqr, q3 + 1.5 * iqr
    valores = {
        "media": serie.mean(),
        "mediana": serie.median(),
        "desviacion": serie.std(),
        "q1": q1,
        "q3": q3,
        "iqr": iqr,
        "limite_inferior": inferior,
        "limite_superior": superior,
        "asimetria": serie.skew(),
    }
    resultado = {"n": int(serie.count())}
    resultado.update({k: round(float(v), 2) for k, v in valores.items()})
    resultado["atipicos"] = int(((serie < inferior) | (serie > superior)).sum())
    return resultado


def atipicos_por_tipo(cdr):
    voz = cdr.loc[(cdr["tipo"] == "VOZ") & (cdr["duracion_seg"] > 0), "duracion_seg"]
    datos = cdr.loc[cdr["tipo"] == "DATOS", "mb"]
    return {
        "VOZ": describir(np.log1p(voz))["atipicos"],
        "DATOS": describir(np.log1p(datos))["atipicos"],
    }


print(describir(cdr.loc[cdr["tipo"] == "DATOS", "mb"]))
print(atipicos_por_tipo(cdr))
```

## Explicación

Ejecuta `describir` sobre las megas sin logaritmo: la asimetría es grande y positiva, y la regla del IQR marca como "atípicas" muchas sesiones perfectamente normales, solo porque la distribución tiene cola larga. Sobre el logaritmo, la forma se parece a una campana y los atípicos que quedan son de verdad inusuales.

Elegir la escala correcta antes de aplicar una regla estadística es una decisión de análisis tan importante como la regla misma.
