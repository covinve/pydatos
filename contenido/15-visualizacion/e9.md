## Enunciado

El análisis exploratorio se repite cada mes con los datos nuevos. Conviértelo en un **script** que genere el reporte completo con un comando. Completa `generar_reporte(carpeta_datos, carpeta_salida)` en `ejercicios/m15-e9/reporte_eda.py` (lee su docstring: define cada indicador).

El script debe crear en la carpeta de salida:

| Archivo | Contenido |
|---|---|
| `registros_por_dia.png` | Línea de CDR válidos por día con su media móvil de 7 días |
| `duracion_llamadas.png` | Histograma de la duración de las llamadas válidas (escala logarítmica) |
| `consumo_por_plan.png` | Cajas de minutos por cliente según el plan |
| `mapa_dia_hora.png` | Mapa de calor día de la semana × hora |
| `reporte.md` | Título, una sección `##` por gráfico con su imagen y una frase con un dato concreto, y una tabla de indicadores |

Comprueba con `python comprobar.py m15-e9` y luego genera el de 10 MB:

```powershell
python ejercicios/m15-e9/reporte_eda.py datos/10MB reportes/eda_10MB
```

Abre `reportes/eda_10MB/reporte.md` en VS Code y pulsa `Ctrl + Shift + V` para ver la vista previa con las imágenes.

## Pista 1

Organiza el código en funciones: una que lee y valida (devuelve el DataFrame de válidos), una por gráfico (cada una recibe los datos y la ruta donde guardar), una que calcula los indicadores y una que escribe el Markdown. `generar_reporte` solo las llama en orden.

Para cada gráfico: crea la figura, dibuja, `fig.savefig(ruta, dpi=120, bbox_inches="tight")` y **`plt.close(fig)`** (si no cierras las figuras, se acumulan en memoria).

## Pista 2

La tabla de indicadores en Markdown:

```python
filas = "\n".join(f"| {nombre} | {valor} |" for nombre, valor in indicadores.items())
tabla = "| Indicador | Valor |\n|---|---|\n" + filas
```

Para la hora pico: `voz_validas["inicio"].dt.hour.value_counts().sort_index().idxmax()` (al ordenar por hora, `idxmax` devuelve la más temprana en un empate).

## Solución

```python reporte_eda.py
"""Ejercicio m15-e9: reporte exploratorio automático.

Uso:
    python ejercicios/m15-e9/reporte_eda.py datos/10MB reportes/eda_10MB
"""

from __future__ import annotations

import sys
from pathlib import Path

import matplotlib

matplotlib.use("Agg")  # genera imágenes sin abrir ventanas

import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
import pandas as pd  # noqa: E402

DIAS = ["Lun", "Mar", "Mié", "Jue", "Vie", "Sáb", "Dom"]


def miles(valor) -> str:
    return f"{valor:,}".replace(",", ".")


def leer_validos(carpeta: Path) -> tuple[pd.DataFrame, pd.DataFrame, int]:
    cdr = pd.read_csv(carpeta / "cdr.csv", sep="|", keep_default_na=False, dtype={"linea_origen": "string"})
    clientes = pd.read_csv(carpeta / "clientes.csv", encoding="latin-1", dtype={"linea": "string"})
    inicio = pd.to_datetime(cdr["fecha_inicio"], format="%Y-%m-%d %H:%M:%S", errors="coerce")
    validos = (~cdr.duplicated("cdr_id") & (cdr["linea_origen"] != "") & cdr["linea_origen"].isin(clientes["linea"])
               & cdr["tipo"].isin(["VOZ", "SMS", "DATOS"]) & (cdr["duracion_seg"] >= 0) & inicio.notna())
    datos = cdr[validos].assign(inicio=inicio[validos])
    datos["minutos"] = np.where(datos["tipo"] == "VOZ", (datos["duracion_seg"] + 59) // 60, 0)
    return datos, clientes, len(cdr)


def guardar(fig, ruta: Path) -> None:
    fig.savefig(ruta, dpi=120, bbox_inches="tight")
    plt.close(fig)


def grafico_diario(datos, ruta) -> None:
    por_dia = datos.set_index("inicio").resample("D").size()
    fig, ax = plt.subplots(figsize=(8, 3))
    ax.plot(por_dia.index, por_dia.values, label="Registros válidos")
    ax.plot(por_dia.index, por_dia.rolling(7).mean().values, linestyle="--", label="Media móvil 7 días")
    ax.set_title("Registros válidos por día")
    ax.set_ylim(bottom=0)
    ax.legend()
    fig.autofmt_xdate()
    guardar(fig, ruta)


def grafico_duracion(voz, ruta) -> None:
    d = voz.loc[voz["duracion_seg"] > 0, "duracion_seg"]
    bordes = np.logspace(np.log10(d.min()), np.log10(d.max()), 31)
    bordes[0], bordes[-1] = d.min(), d.max()
    fig, ax = plt.subplots(figsize=(7, 3))
    ax.hist(d, bins=bordes, color="#0891b2", edgecolor="white")
    ax.set_xscale("log")
    ax.set_title("Duración de las llamadas (segundos, escala logarítmica)")
    guardar(fig, ruta)


def grafico_planes(datos, clientes, ruta) -> pd.DataFrame:
    minutos = datos.groupby("linea_origen")["minutos"].sum()
    tabla = clientes[["linea", "plan_id"]].assign(minutos=lambda t: t["linea"].map(minutos).fillna(0))
    planes = sorted(tabla["plan_id"].unique())
    fig, ax = plt.subplots(figsize=(8, 3.5))
    ax.boxplot([tabla.loc[tabla["plan_id"] == p, "minutos"] for p in planes], tick_labels=planes)
    ax.set_title("Minutos al mes por cliente, según el plan")
    guardar(fig, ruta)
    return tabla


def grafico_mapa(voz, ruta) -> None:
    matriz = pd.crosstab(voz["inicio"].dt.dayofweek, voz["inicio"].dt.hour).reindex(
        index=range(7), columns=range(24), fill_value=0)
    fig, ax = plt.subplots(figsize=(10, 3))
    imagen = ax.imshow(matriz.to_numpy(), aspect="auto", cmap="viridis")
    ax.set_yticks(range(7), DIAS)
    ax.set_xlabel("Hora del día")
    fig.colorbar(imagen, ax=ax, label="Llamadas")
    guardar(fig, ruta)


def indicadores(datos, clientes, total) -> dict:
    voz = datos[datos["tipo"] == "VOZ"]
    return {
        "registros": int(total),
        "validos": int(len(datos)),
        "pct_validos": round(len(datos) / total * 100, 1),
        "minutos": int(voz["minutos"].sum()),
        "gb": round(float(datos.loc[datos["tipo"] == "DATOS", "mb"].sum()) / 1024, 2),
        "hora_pico": int(voz["inicio"].dt.hour.value_counts().sort_index().idxmax()),
        "mediana_duracion_seg": float(voz.loc[voz["duracion_seg"] > 0, "duracion_seg"].median()),
        "clientes_sin_consumo": int((~clientes["linea"].isin(datos["linea_origen"])).sum()),
    }


def escribir_markdown(ind: dict, ruta: Path) -> None:
    tabla = "| Indicador | Valor |\n|---|---|\n" + "\n".join(f"| {k} | {v} |" for k, v in ind.items())
    texto = f"""# Reporte exploratorio de consumo

## Registros por día

![Registros por día](registros_por_dia.png)

Se validaron {miles(ind['validos'])} de {miles(ind['registros'])} registros ({ind['pct_validos']} %).

## Duración de las llamadas

![Duración](duracion_llamadas.png)

La mediana de duración es {ind['mediana_duracion_seg']:.0f} segundos.

## Consumo por plan

![Consumo por plan](consumo_por_plan.png)

{ind['clientes_sin_consumo']} clientes no tuvieron ningún consumo válido.

## Día y hora

![Mapa día y hora](mapa_dia_hora.png)

La hora con más llamadas es las {ind['hora_pico']:02d}:00.

## Indicadores

{tabla}
"""
    ruta.write_text(texto, encoding="utf-8")


def generar_reporte(carpeta_datos: Path, carpeta_salida: Path) -> dict:
    carpeta_salida.mkdir(parents=True, exist_ok=True)
    datos, clientes, total = leer_validos(carpeta_datos)
    voz = datos[datos["tipo"] == "VOZ"]
    grafico_diario(datos, carpeta_salida / "registros_por_dia.png")
    grafico_duracion(voz, carpeta_salida / "duracion_llamadas.png")
    grafico_planes(datos, clientes, carpeta_salida / "consumo_por_plan.png")
    grafico_mapa(voz, carpeta_salida / "mapa_dia_hora.png")
    ind = indicadores(datos, clientes, total)
    escribir_markdown(ind, carpeta_salida / "reporte.md")
    return ind


if __name__ == "__main__":
    print(generar_reporte(Path(sys.argv[1]), Path(sys.argv[2])))
```

## Explicación

Un reporte generado por código tiene tres ventajas sobre uno armado a mano:

- **Es repetible**: el reporte de octubre es el mismo comando con otra carpeta.
- **Es verificable**: las pruebas comparan los indicadores con `esperado.json`; un reporte a mano no tiene esa red de seguridad.
- **Las frases se escriben solas a partir de los datos**: nunca quedan desactualizadas.

`matplotlib.use("Agg")` hace que matplotlib dibuje directamente a archivos, sin intentar abrir ventanas: indispensable cuando el script corre en un servidor o programado por la noche.

Markdown es un formato excelente para reportes: se lee bien como texto, VS Code y GitHub lo muestran con formato, y se puede convertir a HTML o PDF. En el Proyecto 2, tu reporte de conciliación seguirá esta misma estructura.
