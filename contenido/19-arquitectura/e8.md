## Enunciado

El principio de sustitución (la **L** de SOLID) dice que cualquier implementación de un puerto debe poder usarse en lugar de otra. ¿Cómo se garantiza? Con **pruebas de contrato**: una sola batería de pruebas que se ejecuta contra **todas** las implementaciones.

En `ejercicios/m19-e8/repositorios.py` está el puerto `RepositorioClientes`. Implementa:

1. `ClientesEnMemoria`: con un diccionario.
2. `ClientesSQLite(ruta_bd)`: con SQLite (consultas parametrizadas, `fecha_alta` guardada en ISO y devuelta como `date`).
3. `ClientesJSON(ruta)`: con un archivo JSON, escrito de forma **atómica** (temporal + `Path.replace`).
4. `cargar_csv(repo, ruta_csv)`: que funcione con cualquiera de las tres.

`test_contrato.py` usa un *fixture* **parametrizado**: cada prueba del contrato se ejecuta tres veces, una por implementación. Fíjate en la salida de `pytest -v`: `test_guardar_y_obtener[memoria]`, `[sqlite]`, `[json]`.

## Pista 1

SQLite con `INSERT ... ON CONFLICT` para que `guardar` reemplace:

```python
class ClientesSQLite:
    def __init__(self, ruta_bd):
        self.con = sqlite3.connect(ruta_bd)
        self.con.execute("CREATE TABLE IF NOT EXISTS clientes (linea TEXT PRIMARY KEY, nombre TEXT, ...)")

    def guardar(self, c):
        with self.con:
            self.con.execute("INSERT INTO clientes VALUES (?, ?, ?, ?, ?) ON CONFLICT (linea) DO UPDATE SET ...",
                             (c.linea, c.nombre, c.plan_id, c.segmento, c.fecha_alta.isoformat()))
```

Al leer: `Cliente(linea, nombre, plan, segmento, date.fromisoformat(fecha))`.

## Pista 2

`ClientesJSON` puede heredar de `ClientesEnMemoria`: al crearse, carga el archivo (si existe) en el diccionario; cada `guardar` llama al `guardar` del padre y luego escribe todo:

```python
def _escribir(self):
    datos = [{**asdict(c), "fecha_alta": c.fecha_alta.isoformat()} for c in self._clientes.values()]
    temporal = self.ruta.with_suffix(".tmp")
    temporal.write_text(json.dumps(datos, ensure_ascii=False, indent=2), encoding="utf-8")
    temporal.replace(self.ruta)
```

## Solución

```python repositorios.py
"""Ejercicio m19-e8: un puerto, tres adaptadores y una sola batería de pruebas de contrato."""

from __future__ import annotations

import csv
import json
import sqlite3
from dataclasses import asdict, dataclass
from datetime import date
from pathlib import Path
from typing import Protocol


@dataclass(frozen=True)
class Cliente:
    linea: str
    nombre: str
    plan_id: str
    segmento: str
    fecha_alta: date


class ClienteNoEncontrado(LookupError):
    pass


class RepositorioClientes(Protocol):
    def obtener(self, linea: str) -> Cliente:
        """El cliente de esa línea. ClienteNoEncontrado si no existe."""
        ...

    def guardar(self, cliente: Cliente) -> None:
        """Lo agrega o, si la línea ya existe, lo reemplaza."""
        ...

    def del_plan(self, plan_id: str) -> list[Cliente]:
        """Los clientes de ese plan, ordenados por línea ([] si no hay)."""
        ...

    def contar(self) -> int:
        ...


class ClientesEnMemoria:
    def __init__(self):
        self._clientes: dict[str, Cliente] = {}

    def obtener(self, linea):
        try:
            return self._clientes[linea]
        except KeyError:
            raise ClienteNoEncontrado(linea) from None

    def guardar(self, cliente):
        self._clientes[cliente.linea] = cliente

    def del_plan(self, plan_id):
        return sorted((c for c in self._clientes.values() if c.plan_id == plan_id), key=lambda c: c.linea)

    def contar(self):
        return len(self._clientes)


class ClientesSQLite:
    def __init__(self, ruta_bd):
        self.con = sqlite3.connect(ruta_bd)
        with self.con:
            self.con.execute("""
                CREATE TABLE IF NOT EXISTS clientes (
                    linea TEXT PRIMARY KEY, nombre TEXT, plan_id TEXT, segmento TEXT, fecha_alta TEXT
                )
            """)

    @staticmethod
    def _cliente(fila):
        linea, nombre, plan_id, segmento, fecha = fila
        return Cliente(linea, nombre, plan_id, segmento, date.fromisoformat(fecha))

    def obtener(self, linea):
        fila = self.con.execute("SELECT * FROM clientes WHERE linea = ?", (linea,)).fetchone()
        if fila is None:
            raise ClienteNoEncontrado(linea)
        return self._cliente(fila)

    def guardar(self, c):
        with self.con:
            self.con.execute("""
                INSERT INTO clientes VALUES (?, ?, ?, ?, ?)
                ON CONFLICT (linea) DO UPDATE SET nombre = excluded.nombre, plan_id = excluded.plan_id,
                    segmento = excluded.segmento, fecha_alta = excluded.fecha_alta
            """, (c.linea, c.nombre, c.plan_id, c.segmento, c.fecha_alta.isoformat()))

    def del_plan(self, plan_id):
        filas = self.con.execute("SELECT * FROM clientes WHERE plan_id = ? ORDER BY linea", (plan_id,))
        return [self._cliente(f) for f in filas]

    def contar(self):
        return self.con.execute("SELECT COUNT(*) FROM clientes").fetchone()[0]


class ClientesJSON(ClientesEnMemoria):
    def __init__(self, ruta):
        super().__init__()
        self.ruta = Path(ruta)
        if self.ruta.exists():
            for d in json.loads(self.ruta.read_text(encoding="utf-8")):
                super().guardar(Cliente(**{**d, "fecha_alta": date.fromisoformat(d["fecha_alta"])}))

    def guardar(self, cliente):
        super().guardar(cliente)
        datos = [{**asdict(c), "fecha_alta": c.fecha_alta.isoformat()} for c in self._clientes.values()]
        temporal = self.ruta.with_name(self.ruta.name + ".tmp")
        temporal.write_text(json.dumps(datos, ensure_ascii=False, indent=2), encoding="utf-8")
        temporal.replace(self.ruta)


def cargar_csv(repo: RepositorioClientes, ruta_csv: Path) -> int:
    n = 0
    with open(ruta_csv, encoding="latin-1", newline="") as f:
        for c in csv.DictReader(f):
            repo.guardar(Cliente(c["linea"], c["nombre"], c["plan_id"], c["segmento"],
                                 date.fromisoformat(c["fecha_alta"])))
            n += 1
    return n
```

## Explicación

Las pruebas de contrato detectan las diferencias sutiles que rompen la sustitución, justo las que aparecen en producción:

- SQLite devuelve la fecha como **texto** si no la conviertes: `obtener` daría un `Cliente` "casi igual".
- Un JSON escrito con `ensure_ascii=True` guarda `Ñandú`: funciona, pero nadie puede leerlo.
- Un `del_plan` sin `ORDER BY` devuelve un orden "que casi siempre coincide"… hasta que no.

La escritura **atómica** del JSON (temporal + `replace`) evita un desastre clásico: si el programa se cae a mitad de `write_text`, el archivo queda cortado y **todos** los clientes se pierden. `Path.replace` cambia un archivo por otro en una sola operación del sistema operativo: o queda el viejo completo o el nuevo completo.

Un detalle de rendimiento: `ClientesJSON` reescribe todo el archivo en cada `guardar`. Con 40 clientes da igual; con 2 millones, `cargar_csv` sería lentísimo. El contrato no lo prohíbe, pero es una buena razón para que existan varias implementaciones: JSON para configuraciones pequeñas, SQLite o PostgreSQL para volumen.
