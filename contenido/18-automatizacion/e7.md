## Enunciado

En `ejercicios/m18-e7/telco_cli.py` la lógica ya está hecha (`validar_cdr`, `top_lineas`). Te falta la **interfaz**: una CLI con Typer y tres comandos (`resumen`, `validar`, `version`). El comentario al final del archivo describe cada uno con sus argumentos, opciones, validaciones, salidas y códigos.

Reglas de diseño:

- Las validaciones de argumentos las hace **Typer** (`exists=True`, `file_okay`/`dir_okay`, `min=1`, un `Enum` para el tipo), no tu código.
- Las funciones de los comandos son **delgadas**: leen opciones, llaman la lógica, muestran el resultado y deciden el código de salida.

Cuando pase `python comprobar.py m18-e7`, pruébala de verdad:

```powershell
python ejercicios/m18-e7/telco_cli.py --help
python ejercicios/m18-e7/telco_cli.py validar datos/10MB -o salidas/hoy
$LASTEXITCODE
```

## Pista 1

Un `Enum` de texto da a Typer la lista de valores válidos:

```python
from enum import Enum

class Tipo(str, Enum):
    VOZ = "VOZ"
    SMS = "SMS"
    DATOS = "DATOS"

@app.command()
def resumen(
    archivo: Annotated[Path, typer.Argument(exists=True, file_okay=True, dir_okay=False, help="...")],
    tipo: Annotated[Tipo | None, typer.Option(help="...")] = None,
    top: Annotated[int, typer.Option(min=1, help="...")] = 5,

    for linea, n in top_lineas(archivo, tipo.value if tipo else None, top):
        typer.echo(f"{linea}: {n}")
```

## Pista 2

Para la opción con nombre corto: `typer.Option("--salida", "-o", help="...")`. Para terminar con código 1: `raise typer.Exit(code=1)`, **después** de escribir los archivos.

Al escribir `rechazos.csv` con `csv.writer`, abre el archivo con `newline=""` para no tener líneas en blanco de más en Windows.

## Solución

```python telco_cli.py
"""Ejercicio m18-e7: la CLI del equipo de facturación, con Typer.

Uso (cuando la termines):
    python ejercicios/m18-e7/telco_cli.py --help
    python ejercicios/m18-e7/telco_cli.py resumen datos/10MB/cdr.csv --tipo VOZ --top 3
    python ejercicios/m18-e7/telco_cli.py validar datos/10MB --salida salidas/hoy --max-rechazo 2.5
"""

from __future__ import annotations

import csv
import json
from collections import Counter
from datetime import datetime
from enum import Enum
from pathlib import Path
from typing import Annotated, Optional

import typer

__version__ = "1.0.0"
ERRORES = ["duplicado", "duracion_negativa", "fecha_invalida", "linea_vacia", "linea_desconocida", "tipo_desconocido"]


# ---------------------------------------------------------------------------
# Lógica (ya está hecha: la CLI solo la usa)
# ---------------------------------------------------------------------------

def motivo(fila: dict, vistos: set, clientes: set) -> str:
    if fila["cdr_id"] in vistos:
        return "duplicado"
    vistos.add(fila["cdr_id"])
    if not fila["linea_origen"]:
        return "linea_vacia"
    if fila["linea_origen"] not in clientes:
        return "linea_desconocida"
    if fila["tipo"] not in ("VOZ", "SMS", "DATOS"):
        return "tipo_desconocido"
    if int(fila["duracion_seg"]) < 0:
        return "duracion_negativa"
    try:
        datetime.strptime(fila["fecha_inicio"], "%Y-%m-%d %H:%M:%S")
    except ValueError:
        return "fecha_invalida"
    return "valido"


def validar_cdr(carpeta: Path) -> tuple[dict, list[tuple[str, str]]]:
    """Devuelve (resumen, rechazos): resumen con registros_totales, registros_validos y errores;
    rechazos como lista de (cdr_id, motivo)."""
    with open(carpeta / "clientes.csv", encoding="latin-1", newline="") as f:
        clientes = {c["linea"] for c in csv.DictReader(f)}
    vistos: set = set()
    conteo: Counter = Counter()
    rechazos = []
    with open(carpeta / "cdr.csv", encoding="utf-8", newline="") as f:
        for fila in csv.DictReader(f, delimiter="|"):
            m = motivo(fila, vistos, clientes)
            conteo[m] += 1
            if m != "valido":
                rechazos.append((fila["cdr_id"], m))
    resumen = {
        "registros_totales": sum(conteo.values()),
        "registros_validos": conteo["valido"],
        "errores": {e: conteo[e] for e in ERRORES},
    }
    return resumen, rechazos


def top_lineas(archivo: Path, tipo: str | None, top: int) -> list[tuple[str, int]]:
    conteo: Counter = Counter()
    with open(archivo, encoding="utf-8", newline="") as f:
        for fila in csv.DictReader(f, delimiter="|"):
            if fila["linea_origen"] and (tipo is None or fila["tipo"] == tipo):
                conteo[fila["linea_origen"]] += 1
    return sorted(conteo.items(), key=lambda par: (-par[1], par[0]))[:top]


# ---------------------------------------------------------------------------
# La CLI
# ---------------------------------------------------------------------------

class Tipo(str, Enum):
    VOZ = "VOZ"
    SMS = "SMS"
    DATOS = "DATOS"


app = typer.Typer(help="Herramientas del equipo de facturación telco.", no_args_is_help=True)


@app.command()
def resumen(
    archivo: Annotated[Path, typer.Argument(exists=True, file_okay=True, dir_okay=False, help="Ruta del cdr.csv")],
    tipo: Annotated[Optional[Tipo], typer.Option(help="Solo este tipo de CDR")] = None,
    top: Annotated[int, typer.Option(min=1, help="Cuántas líneas mostrar")] = 5,
):
    """Líneas de origen con más registros."""
    for linea, n in top_lineas(archivo, tipo.value if tipo else None, top):
        typer.echo(f"{linea}: {n}")


@app.command()
def validar(
    carpeta: Annotated[Path, typer.Argument(exists=True, file_okay=False, dir_okay=True,
                                            help="Carpeta con cdr.csv y clientes.csv")],
    salida: Annotated[Path, typer.Option("--salida", "-o", help="Carpeta de resultados")] = Path("salidas"),
    max_rechazo: Annotated[float, typer.Option(help="Porcentaje máximo de rechazos aceptable")] = 5.0,
):
    """Valida los CDR y escribe resumen.json y rechazos.csv."""
    resumen_validacion, rechazos = validar_cdr(carpeta)
    salida.mkdir(parents=True, exist_ok=True)
    (salida / "resumen.json").write_text(json.dumps(resumen_validacion, indent=2, ensure_ascii=False), encoding="utf-8")
    with open(salida / "rechazos.csv", "w", encoding="utf-8", newline="") as f:
        escritor = csv.writer(f)
        escritor.writerow(["cdr_id", "motivo"])
        escritor.writerows(rechazos)

    total = resumen_validacion["registros_totales"]
    validos = resumen_validacion["registros_validos"]
    porcentaje = (total - validos) / total * 100 if total else 0.0
    typer.echo(f"Válidos: {validos} de {total} ({porcentaje:.2f} % rechazados)")
    if porcentaje > max_rechazo:
        typer.echo(f"ALERTA: el {porcentaje:.2f} % de rechazos supera el máximo de {max_rechazo} %")
        raise typer.Exit(code=1)


@app.command()
def version():
    """Muestra la versión."""
    typer.echo(f"telco-cli {__version__}")


if __name__ == "__main__":
    app()
```

## Explicación

Compara cuánto código de validación de argumentos escribiste: **ninguno**. Que el archivo exista, que sea archivo y no carpeta, que `--top` sea un entero mayor que 0, que `--tipo` sea uno de tres valores: todo sale de las anotaciones. Y la ayuda se arma sola con los `help=` y los docstrings.

El código de salida 1 cuando hay demasiados rechazos convierte a esta CLI en una pieza de **automatización**: el Programador de tareas, otro script o GitHub Actions pueden decidir qué hacer ("enviar alerta", "no publicar el reporte") sin leer el texto.

Fíjate en el orden dentro de `validar`: primero se escriben los archivos y **después** se decide el código. Si hay una alerta, quien la investigue necesita justamente esos archivos.

Para tenerla como comando `telco` en tu entorno, agrégala a un `pyproject.toml` con `[project.scripts] telco = "telco_cli:app"` e instala con `pip install -e .`, como en el proyecto 2.
