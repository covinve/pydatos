## Enunciado

Escribe un pequeño "resolvedor" de versiones como el que usa pip.

1. `version(texto)` → tupla de enteros: `"2.2.3"` → `(2, 2, 3)`. Para comparar, `"2.2"` equivale a `"2.2.0"`: rellena con ceros hasta 3 partes.
2. `cumple(v, condicion)` → `True` si la versión `v` (texto) cumple **una** condición como `">=2.2"`, `"<3"`, `"==1.1.3"`, `"!=2.1.1"`, `">2"`, `"<=1.5"` o `"~=2.2.0"`.
   - `~=X.Y.Z` significa `>=X.Y.Z` y `==X.Y.*` (`<X.(Y+1)`); `~=X.Y` significa `>=X.Y` y `<(X+1)`.
3. `cumple_todas(v, especificador)` → para especificadores con comas (`">=2.2,<3,!=2.2.1"`); el vacío `""` acepta todo.
4. `elegir(disponibles, especificador)` → la versión **más alta** de la lista que cumple, o `None`.

## Pruebas

```python
PANDAS = ["1.5.3", "2.0.3", "2.1.4", "2.2.0", "2.2.1", "2.2.2", "2.2.3", "3.0.0", "3.0.1", "10.0.0"]


def prueba_version():
    """Convierte y rellena"""
    requiere("version")
    igual(version("2.2.3"), (2, 2, 3))
    igual(version("2.2"), (2, 2, 0))
    igual(version("3"), (3, 0, 0))
    assert version("10.0.0") > version("9.9.9"), "Compara como números, no como texto"


def prueba_cumple():
    """Cada operador"""
    requiere("cumple")
    casos = [("2.2.3", ">=2.2", True), ("2.1.9", ">=2.2", False), ("2.9.9", "<3", True), ("3.0.0", "<3", False),
             ("1.1.3", "==1.1.3", True), ("1.1", "==1.1.0", True), ("2.1.1", "!=2.1.1", False), ("2.1.2", "!=2.1.1", True),
             ("2.0.1", ">2", True), ("2.0.0", ">2", False), ("1.5", "<=1.5", True), ("1.5.1", "<=1.5", False)]
    for v, c, esperado in casos:
        igual(cumple(v, c), esperado, f"{v} {c}")


def prueba_compatible():
    """El operador ~="""
    igual([v for v in PANDAS if cumple(v, "~=2.2.0")], ["2.2.0", "2.2.1", "2.2.2", "2.2.3"])
    igual([v for v in PANDAS if cumple(v, "~=2.1")], ["2.1.4", "2.2.0", "2.2.1", "2.2.2", "2.2.3"])


def prueba_todas():
    """Varias condiciones"""
    requiere("cumple_todas")
    igual(cumple_todas("2.2.3", ">=2.2,<3,!=2.2.1"), True)
    igual(cumple_todas("2.2.1", ">=2.2,<3,!=2.2.1"), False)
    igual(cumple_todas("0.1", ""), True)
    igual(cumple_todas("2.2.0", " >= 2.2 , < 3 "), True, "Tolera espacios")


def prueba_elegir():
    """La más alta que cumple"""
    requiere("elegir")
    igual(elegir(PANDAS, ">=2.2,<3"), "2.2.3")
    igual(elegir(PANDAS, ""), "10.0.0")
    igual(elegir(PANDAS, ">=2.2"), "10.0.0", "¡Sin límite superior se instala la 10!")
    igual(elegir(PANDAS, ">=11"), None)
    igual(elegir(PANDAS, "~=2.1.0"), "2.1.4")
```

## Pista 2

Para `~=`, el límite superior depende de cuántas partes tiene el objetivo **como se escribió**:

```python
partes = [int(p) for p in objetivo.split(".")]     # "2.2.0" → [2, 2, 0]; "2.1" → [2, 1]
tope = partes[:-1]                                  # quita la última: [2, 2] o [2]
tope[-1] += 1                                       # [2, 3] o [3]
return version(v) >= version(objetivo) and version(v) < version(".".join(map(str, tope)))
```

## Solución

```python
OPERADORES = ("~=", "==", "!=", ">=", "<=", ">", "<")


def version(texto):
    partes = [int(p) for p in texto.strip().split(".")]
    return tuple((partes + [0, 0, 0])[:3])


def cumple(v, condicion):
    condicion = condicion.strip()
    operador = next(op for op in OPERADORES if condicion.startswith(op))
    objetivo = condicion[len(operador):].strip()
    actual, meta = version(v), version(objetivo)
    if operador == "~=":
        tope = [int(p) for p in objetivo.split(".")][:-1]
        tope[-1] += 1
        return actual >= meta and actual < version(".".join(map(str, tope)))
    return {
        "==": actual == meta, "!=": actual != meta, ">=": actual >= meta,
        "<=": actual <= meta, ">": actual > meta, "<": actual < meta,
    }[operador]


def cumple_todas(v, especificador):
    return all(cumple(v, c) for c in especificador.split(",") if c.strip())


def elegir(disponibles, especificador):
    candidatas = [v for v in disponibles if cumple_todas(v, especificador)]
    return max(candidatas, key=version, default=None)


PANDAS = ["1.5.3", "2.0.3", "2.1.4", "2.2.0", "2.2.1", "2.2.2", "2.2.3", "3.0.0", "3.0.1"]
print(elegir(PANDAS, ">=2.2,<3"))
print(elegir(PANDAS, "~=2.1.0"))
```

## Explicación

Esta es, en miniatura, la decisión que pip toma por cada librería al instalar. La prueba `">=2.2"` → `"10.0.0"` muestra por qué un especificador sin límite superior es peligroso en una aplicación: el día que salga pandas 3 (o 10), un `pip install` nuevo la instalará, aunque tu código nunca se probó con ella.

Las versiones se comparan como **tuplas de números**, no como texto: como texto, `"10.0.0" < "9.9.9"` (porque `"1" < "9"`), un error clásico al ordenar versiones con `sorted()` a secas.

La versión real (PEP 440) tiene más casos: prelanzamientos (`3.0.0rc1`), versiones de desarrollo, comodines (`==2.2.*`), épocas. La librería `packaging` (`from packaging.specifiers import SpecifierSet`) los maneja todos; úsala cuando lo necesites en serio.
