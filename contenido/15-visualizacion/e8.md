```python
import numpy as np
import pandas as pd

CDR = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"}, keep_default_na=False)
CLI = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})


def ref():
    v = CDR[CDR["duracion_seg"] >= 0]
    m = ((v["duracion_seg"] + 59) // 60).where(v["tipo"] == "VOZ", 0).groupby(v["linea_origen"]).sum()
    g = (v["mb"].where(v["tipo"] == "DATOS", 0) / 1024).groupby(v["linea_origen"]).sum()
    t = CLI.set_index("linea")[["segmento"]].copy()
    t["minutos"] = m.reindex(t.index, fill_value=0)
    t["gb"] = g.reindex(t.index, fill_value=0)
    return t


def prueba_tabla():
    """Todos los clientes con su consumo"""
    t = consumo_cliente(CDR, CLI)
    r = ref()
    igual(len(t), len(CLI))
    igual(int(t["minutos"].sum()), int(r["minutos"].sum()))
    cerca(float(t["gb"].sum()), float(r["gb"].sum()), 0.001)


def prueba_correlaciones():
    """Pearson y Spearman"""
    r = ref()
    c = correlaciones(consumo_cliente(CDR, CLI))
    igual(c, {"pearson": round(float(r["minutos"].corr(r["gb"])), 3),
              "spearman": round(float(r["minutos"].corr(r["gb"], method="spearman")), 3)})


def prueba_grafico():
    """Un grupo de puntos por segmento, recta y título con Spearman"""
    t = consumo_cliente(CDR, CLI)
    ax = dispersion(t).axes[0]
    igual(len(ax.collections), t["segmento"].nunique())
    igual(len(ax.lines), 1)
    rho = correlaciones(t)["spearman"]
    igual(ax.get_title(), f"Minutos vs. GB por cliente (Spearman = {rho:.2f})")
    etiquetas = [h.get_label() for h in ax.get_legend().legend_handles] if ax.get_legend() else []
    assert set(t["segmento"].unique()) <= set(etiquetas), "La leyenda debe mostrar los segmentos"
```

## Solución

```python
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

cdr = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"}, keep_default_na=False)
clientes = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})


def consumo_cliente(cdr, clientes):
    v = cdr[cdr["duracion_seg"] >= 0]
    minutos = ((v["duracion_seg"] + 59) // 60).where(v["tipo"] == "VOZ", 0).groupby(v["linea_origen"]).sum()
    gb = (v["mb"].where(v["tipo"] == "DATOS", 0) / 1024).groupby(v["linea_origen"]).sum()
    tabla = clientes.set_index("linea")[["segmento"]].copy()
    tabla["minutos"] = minutos.reindex(tabla.index, fill_value=0)
    tabla["gb"] = gb.reindex(tabla.index, fill_value=0)
    return tabla


def correlaciones(tabla):
    return {
        "pearson": round(float(tabla["minutos"].corr(tabla["gb"])), 3),
        "spearman": round(float(tabla["minutos"].corr(tabla["gb"], method="spearman")), 3),
    }


def dispersion(tabla):
    fig, ax = plt.subplots(figsize=(7, 4.5))
    for segmento, grupo in tabla.groupby("segmento"):
        ax.scatter(grupo["minutos"], grupo["gb"], label=segmento, alpha=0.8)
    pendiente, intercepto = np.polyfit(tabla["minutos"], tabla["gb"], 1)
    x = np.linspace(tabla["minutos"].min(), tabla["minutos"].max(), 50)
    ax.plot(x, pendiente * x + intercepto, color="#334155", linestyle="--")
    rho = correlaciones(tabla)["spearman"]
    ax.set_title(f"Minutos vs. GB por cliente (Spearman = {rho:.2f})")
    ax.set_xlabel("Minutos de voz")
    ax.set_ylabel("GB de datos")
    ax.legend(title="Segmento")
    return fig


t = consumo_cliente(cdr, clientes)
print(correlaciones(t))
fig = dispersion(t)
```

## Explicación

Colorear por segmento revela algo que la correlación global esconde: los clientes empresariales pueden formar una nube aparte, más arriba y a la derecha. Gran parte de la correlación general puede deberse a esa diferencia entre grupos (ser empresarial aumenta a la vez los minutos y los datos), no a que "hablar" y "navegar" estén ligados en cada cliente. Por eso conviene mirar el gráfico y no solo el número.

Pearson mide relación **lineal** y es sensible a los clientes extremos; Spearman mide si el **orden** coincide y es robusto. Cuando difieren mucho, suele haber pocos valores extremos dominando el resultado.

Fíjate que el título se construye con el mismo valor que calcula `correlaciones`: un gráfico cuyo título se genera de los datos nunca queda desactualizado.
