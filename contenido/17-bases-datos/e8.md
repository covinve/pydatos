## Enunciado

En el ejercicio 6 sabías el plan **actual** de cada cliente, pero no qué plan tenía en marzo. Para refacturar o atender un reclamo de un mes pasado, necesitas el **historial**.

En `ejercicios/m17-e8/clientes_orm.py` los modelos `Plan`, `Cliente` e `HistorialPlan` ya están definidos con el ORM de SQLAlchemy. Léelos con calma y completa las cuatro funciones:

1. `cargar_catalogos(sesion, carpeta)` → crea planes, clientes y el historial inicial de cada cliente.
2. `cambiar_plan(sesion, linea, plan_id, fecha)` → cierra el periodo vigente y abre uno nuevo, con validaciones (lee el docstring).
3. `plan_vigente(sesion, linea, dia)` → qué plan tenía la línea ese día.
4. `resumen_por_segmento(sesion)` → clientes y suma de cargos fijos por segmento, en **una** consulta.

Las pruebas usan una base SQLite en memoria; el mismo código funcionaría en PostgreSQL cambiando solo la URL del engine.

## Solución

```python clientes_orm.py
"""Ejercicio m17-e8: clientes, planes e historial de cambios de plan con el ORM de SQLAlchemy.

Los modelos ya están definidos: tu trabajo son las cuatro funciones del final.
"""

from __future__ import annotations

import csv
from datetime import date
from pathlib import Path

from sqlalchemy import Date, ForeignKey, String, func, select
from sqlalchemy.orm import DeclarativeBase, Mapped, Session, mapped_column, relationship


class Base(DeclarativeBase):
    pass


class Plan(Base):
    __tablename__ = "planes"

    plan_id: Mapped[str] = mapped_column(String(10), primary_key=True)
    nombre: Mapped[str]
    segmento: Mapped[str]
    cargo_fijo: Mapped[int]

    clientes: Mapped[list[Cliente]] = relationship(back_populates="plan")


class Cliente(Base):
    __tablename__ = "clientes"

    linea: Mapped[str] = mapped_column(String(10), primary_key=True)
    cliente_id: Mapped[str]
    nombre: Mapped[str]
    segmento: Mapped[str]
    fecha_alta: Mapped[date] = mapped_column(Date)
    plan_id: Mapped[str] = mapped_column(ForeignKey("planes.plan_id"))

    plan: Mapped[Plan] = relationship(back_populates="clientes")
    historial: Mapped[list[HistorialPlan]] = relationship(back_populates="cliente", order_by="HistorialPlan.desde")


class HistorialPlan(Base):
    """Qué plan tuvo cada línea y desde/hasta cuándo (hasta = None: vigente)."""

    __tablename__ = "historial_planes"

    id: Mapped[int] = mapped_column(primary_key=True)
    linea: Mapped[str] = mapped_column(ForeignKey("clientes.linea"))
    plan_id: Mapped[str] = mapped_column(ForeignKey("planes.plan_id"))
    desde: Mapped[date] = mapped_column(Date)
    hasta: Mapped[date | None] = mapped_column(Date)

    cliente: Mapped[Cliente] = relationship(back_populates="historial")


# ---------------------------------------------------------------------------
# Tu trabajo
# ---------------------------------------------------------------------------

def cargar_catalogos(sesion: Session, carpeta: Path) -> tuple[int, int]:
    carpeta = Path(carpeta)
    with open(carpeta / "planes.csv", encoding="utf-8", newline="") as f:
        planes = {p["plan_id"]: Plan(plan_id=p["plan_id"], nombre=p["nombre"], segmento=p["segmento"],
                                     cargo_fijo=int(p["cargo_fijo"]))
                  for p in csv.DictReader(f)}
    sesion.add_all(planes.values())
    clientes = 0
    with open(carpeta / "clientes.csv", encoding="latin-1", newline="") as f:
        for c in csv.DictReader(f):
            cliente = Cliente(linea=c["linea"], cliente_id=c["cliente_id"], nombre=c["nombre"],
                              segmento=c["segmento"], fecha_alta=date.fromisoformat(c["fecha_alta"]),
                              plan=planes[c["plan_id"]])
            cliente.historial.append(HistorialPlan(plan_id=c["plan_id"], desde=cliente.fecha_alta))
            sesion.add(cliente)
            clientes += 1
    sesion.commit()
    return len(planes), clientes


def cambiar_plan(sesion: Session, linea: str, plan_id: str, fecha: date) -> None:
    cliente = sesion.get(Cliente, linea)
    if cliente is None:
        raise LookupError(f"La línea {linea} no existe")
    plan = sesion.get(Plan, plan_id)
    if plan is None:
        raise LookupError(f"El plan {plan_id} no existe")
    if cliente.plan_id == plan_id:
        raise ValueError(f"La línea {linea} ya tiene el plan {plan_id}")
    vigente = next(h for h in cliente.historial if h.hasta is None)
    if fecha <= vigente.desde:
        raise ValueError(f"La fecha {fecha} debe ser posterior a {vigente.desde}")
    vigente.hasta = fecha
    cliente.historial.append(HistorialPlan(plan_id=plan_id, desde=fecha))
    cliente.plan = plan
    sesion.commit()


def plan_vigente(sesion: Session, linea: str, dia: date) -> str | None:
    return sesion.scalar(
        select(HistorialPlan.plan_id).where(
            HistorialPlan.linea == linea,
            HistorialPlan.desde <= dia,
            HistorialPlan.hasta.is_(None) | (HistorialPlan.hasta > dia),
        )
    )


def resumen_por_segmento(sesion: Session) -> list[tuple[str, int, int]]:
    consulta = (
        select(Cliente.segmento, func.count(Cliente.linea), func.sum(Plan.cargo_fijo))
        .join(Cliente.plan)
        .group_by(Cliente.segmento)
        .order_by(Cliente.segmento)
    )
    return [tuple(fila) for fila in sesion.execute(consulta)]
```

## Explicación

El historial con `desde`/`hasta` es el patrón estándar para datos que cambian en el tiempo (en bodegas de datos se llama **dimensión lenta tipo 2**). Dos convenciones lo hacen fácil de consultar:

- El periodo es **semiabierto**: `desde <= día < hasta`. Así, el día del cambio pertenece al plan nuevo y nunca hay dos planes vigentes a la vez.
- `hasta = NULL` marca el periodo vigente.

Con el ORM, las relaciones hacen el código casi como lenguaje natural: `cliente.historial.append(...)`, `cliente.plan = plan`. SQLAlchemy calcula qué `INSERT` y `UPDATE` hacer y los envía juntos al `commit()`, en una transacción.

`cambiar_plan` valida **todo antes de modificar nada**: si fallara a la mitad, la sesión quedaría con cambios pendientes. Si una validación solo se puede hacer después de modificar, usa `sesion.rollback()` antes de lanzar el error.

Por último, la prueba del resumen cuenta las consultas enviadas a la base. Un error clásico con ORM es el problema **N+1**: recorrer los clientes y acceder a `cliente.plan` en un ciclo dispara una consulta por cliente. Con 2 millones de clientes, 2 millones de consultas.
