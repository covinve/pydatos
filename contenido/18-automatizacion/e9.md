## Enunciado

Arma el proceso diario completo, listo para dejarlo en el Programador de tareas: **bloqueo**, **registro** en archivo, **bandeja** de entrada, **resumen** en JSON y **códigos de salida** que expliquen qué pasó.

En `ejercicios/m18-e9/tarea_diaria.py` completa:

1. `Bloqueo`: el archivo de bloqueo de la lección 4, más la detección de **bloqueos huérfanos** (más viejos que `vencimiento_horas`).
2. `configurar_registro(carpeta_logs)`: un registro con rotación diaria, sin duplicar manejadores.
3. `ejecutar(base, ahora)`: el proceso, con los pasos y códigos del docstring.
4. `comando_programador(python, script, base, hora)`: el texto del comando `schtasks` para crear la tarea.

Cuando pase `python comprobar.py m18-e9`, prográmalo de verdad (sin admin) con el comando que genera tu función, ejecútalo con `schtasks /Run` y revisa el log y el "Resultado de la última ejecución".

## Pista 2

Para no duplicar manejadores, cierra y quita los anteriores antes de agregar el nuevo:

```python
logger = logging.getLogger(LOGGER)
for anterior in list(logger.handlers):
    anterior.close()
    logger.removeHandler(anterior)
```

En `ejecutar`, el `try/except RuntimeError` va **alrededor** del `with Bloqueo(...)`; y dentro, un `try/except Exception` por archivo con `logger.exception(...)`.

## Solución

```python tarea_diaria.py
"""Ejercicio m18-e9: una tarea diaria lista para el Programador de tareas.

Uso:
    python ejercicios/m18-e9/tarea_diaria.py D:/bandeja_cdr
"""

from __future__ import annotations

import json
import logging
import os
import re
import shutil
import sys
import time
from datetime import datetime
from logging.handlers import TimedRotatingFileHandler
from pathlib import Path

LOGGER = "tarea_diaria"
PATRON = re.compile(r"cdr_\d{8}\.csv")

# Códigos de salida
OK, CON_ERRORES, EN_CURSO, SIN_ARCHIVOS = 0, 1, 3, 4


class Bloqueo:
    """Archivo de bloqueo para que no corran dos copias a la vez.

    - __enter__: crea `ruta` de forma atómica (os.O_CREAT | os.O_EXCL) y escribe dentro el PID y la hora ISO.
      Si ya existe y su fecha de modificación tiene menos de `vencimiento_horas`, lanza RuntimeError.
      Si es más viejo (un bloqueo huérfano), lo borra, registra un WARNING y lo toma.
    - __exit__: lo borra (siempre, aunque haya habido una excepción).
    """

    def __init__(self, ruta: Path, vencimiento_horas: float = 6):
        self.ruta = Path(ruta)
        self.vencimiento_horas = vencimiento_horas

    def __enter__(self):
        if self.ruta.exists():
            edad_horas = (time.time() - self.ruta.stat().st_mtime) / 3600
            if edad_horas < self.vencimiento_horas:
                raise RuntimeError(f"Ya hay un proceso en curso ({self.ruta})")
            logging.getLogger(LOGGER).warning("Bloqueo huérfano de %.1f h en %s: lo reemplazo", edad_horas, self.ruta)
            self.ruta.unlink()
        try:
            descriptor = os.open(self.ruta, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
        except FileExistsError:
            raise RuntimeError(f"Ya hay un proceso en curso ({self.ruta})") from None
        with os.fdopen(descriptor, "w", encoding="utf-8") as f:
            f.write(f"{os.getpid()} {datetime.now().isoformat()}")
        return self

    def __exit__(self, *error):
        self.ruta.unlink(missing_ok=True)


def configurar_registro(carpeta_logs: Path) -> logging.Logger:
    carpeta_logs = Path(carpeta_logs)
    carpeta_logs.mkdir(parents=True, exist_ok=True)
    logger = logging.getLogger(LOGGER)
    for anterior in list(logger.handlers):
        anterior.close()
        logger.removeHandler(anterior)
    manejador = TimedRotatingFileHandler(carpeta_logs / "tarea.log", when="midnight", backupCount=30, encoding="utf-8")
    manejador.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
    logger.addHandler(manejador)
    logger.setLevel(logging.INFO)
    return logger


def _leer(archivo: Path) -> tuple[int, int]:
    lineas = [l for l in archivo.read_text(encoding="utf-8").splitlines() if l]
    posicion = lineas[0].split("|").index("duracion_seg")
    return len(lineas) - 1, sum(int(l.split("|")[posicion]) for l in lineas[1:])


def ejecutar(base: Path, ahora: datetime | None = None) -> int:
    base = Path(base)
    ahora = ahora or datetime.now()
    logger = configurar_registro(base / "logs")
    inicio = time.perf_counter()
    logger.info("Inicio en %s", base)
    try:
        with Bloqueo(base / "tarea.lock"):
            return _procesar(base, ahora, logger, inicio)
    except RuntimeError as e:
        logger.error("No se ejecuta: %s", e)
        return EN_CURSO


def _procesar(base: Path, ahora: datetime, logger: logging.Logger, inicio: float) -> int:
    archivos = sorted(a for a in (base / "entrada").iterdir() if a.is_file() and PATRON.fullmatch(a.name))
    if not archivos:
        logger.warning("No hay archivos para procesar")
        return SIN_ARCHIVOS
    for carpeta in ("procesados", "errores", "salidas"):
        (base / carpeta).mkdir(exist_ok=True)
    resumen = {"fecha": ahora.date().isoformat(), "procesados": [], "errores": []}
    for archivo in archivos:
        try:
            filas, segundos = _leer(archivo)
        except Exception:
            logger.exception("Falló %s: se mueve a errores/", archivo.name)
            shutil.move(archivo, base / "errores" / archivo.name)
            resumen["errores"].append(archivo.name)
            continue
        shutil.move(archivo, base / "procesados" / archivo.name)
        resumen["procesados"].append({"archivo": archivo.name, "filas": filas, "segundos": segundos})
        logger.info("Procesado %s: %d filas", archivo.name, filas)
    salida = base / "salidas" / f"resumen_{ahora:%Y%m%d}.json"
    salida.write_text(json.dumps(resumen, indent=2, ensure_ascii=False), encoding="utf-8")
    logger.info("Fin: %d procesados, %d con error, %.2f s",
                len(resumen["procesados"]), len(resumen["errores"]), time.perf_counter() - inicio)
    return CON_ERRORES if resumen["errores"] else OK


def comando_programador(python: Path, script: Path, base: Path, hora: str = "06:00") -> str:
    accion = f'\\"{Path(python).resolve()}\\" \\"{Path(script).resolve()}\\" \\"{Path(base).resolve()}\\"'
    return f'schtasks /Create /TN "Telco\\TareaDiaria" /SC DAILY /ST {hora} /TR "{accion}" /F'


if __name__ == "__main__":
    sys.exit(ejecutar(Path(sys.argv[1])))
```

## Explicación

Cada pieza responde a una pregunta que alguien te hará algún día:

| Pregunta | Respuesta en el código |
|---|---|
| "¿Corrió hoy?" | El log con `Inicio` y `Fin`, y el código de salida en el Programador |
| "¿Por qué falló el archivo del martes?" | `logger.exception` guardó el *traceback* completo, y el archivo está en `errores/` |
| "¿Se procesó dos veces?" | No: el bloqueo impide ejecuciones simultáneas y la bandeja mueve lo procesado |
| "Se fue la luz a las 6:05, ¿y ahora?" | El bloqueo huérfano vence solo; lo pendiente sigue en `entrada/` |
| "¿Cuántos registros hubo el 3 de septiembre?" | `salidas/resumen_20260903.json` |

Los códigos de salida distintos (`3` en curso, `4` sin archivos) no son un detalle: "no llegó el archivo" y "el proceso se cayó" requieren acciones distintas (llamar al proveedor del archivo o revisar el programa), y otro script puede decidir según el número.

`ejecutar` recibe `ahora` como parámetro por la misma razón que `dormir` en el ejercicio 5: el tiempo es una dependencia que las pruebas necesitan controlar.

Y la última línea, `sys.exit(ejecutar(...))`, es la que le entrega el código al Programador de tareas: sin ella, la tarea siempre diría "0: correcto".
