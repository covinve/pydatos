## Solución

```python informe.py
"""Ejercicio m23-e9: el informe mensual de aseguramiento de ingresos.

Uso:
    python ejercicios/m23-e9/informe.py datos/100MB salidas/informe_202609
"""

from __future__ import annotations

import json
import sys
from pathlib import Path

import matplotlib

matplotlib.use("Agg")                      # sin ventanas: funciona en tareas programadas y en Docker
import matplotlib.pyplot as plt  # noqa: E402
import pandas as pd  # noqa: E402

import calculo  # noqa: E402


def pesos(valor: int) -> str:
    signo = "-" if valor < 0 else ""
    return f"{signo}${abs(int(valor)):,}".replace(",", ".")


def tabla(encabezados: list[str], filas: list[list]) -> str:
    lineas = ["| " + " | ".join(encabezados) + " |", "|" + "---|" * len(encabezados)]
    lineas += ["| " + " | ".join(str(x) for x in fila) + " |" for fila in filas]
    return "\n".join(lineas)


def generar_informe(carpeta: Path, salida: Path) -> dict:
    salida = Path(salida)
    salida.mkdir(parents=True, exist_ok=True)
    correctas = calculo.facturas_correctas(carpeta)[["linea", "total"]].rename(columns={"total": "correcto"})
    emitidas = calculo.facturas_emitidas(carpeta)
    c = emitidas.merge(correctas, on="linea")
    c["facturado"] = c["total"]
    c["diferencia"] = c["facturado"] - c["correcto"]

    con_diferencia = c[c["diferencia"] != 0]
    diferencias = con_diferencia.reindex(con_diferencia["diferencia"].abs().sort_values(ascending=False).index)
    diferencias = diferencias[["factura_id", "linea", "plan_id", "facturado", "correcto", "diferencia"]]
    diferencias.to_csv(salida / "diferencias.csv", index=False, encoding="utf-8")

    d = c["diferencia"]
    resumen = {
        "periodo": str(emitidas["periodo"].iloc[0]),
        "facturas": int(len(c)),
        "total_facturado": int(c["facturado"].sum()),
        "total_correcto": int(c["correcto"].sum()),
        "diferencia_total": int(d.sum()),
        "facturas_con_diferencia": int((d != 0).sum()),
        "perdida": int(-d[d < 0].sum()),
        "cobro_de_mas": int(d[d > 0].sum()),
    }
    (salida / "resumen.json").write_text(json.dumps(resumen, indent=2, ensure_ascii=False), encoding="utf-8")

    por_plan = c.groupby("plan_id")["diferencia"].sum().sort_index()
    fig, ax = plt.subplots(figsize=(8, 4))
    ax.bar(por_plan.index, por_plan.to_numpy(), color=["#c0392b" if v < 0 else "#2e86c1" for v in por_plan])
    ax.axhline(0, color="#555", linewidth=0.8)
    ax.set_title(f"Diferencia facturado − correcto por plan · {resumen['periodo']}")
    ax.set_ylabel("Pesos")
    fig.savefig(salida / "diferencias.png", dpi=120, bbox_inches="tight")
    plt.close(fig)

    indicadores = [
        ["Facturas", f"{resumen['facturas']:,}".replace(",", ".")],
        ["Total facturado", pesos(resumen["total_facturado"])],
        ["Total correcto", pesos(resumen["total_correcto"])],
        ["Diferencia total", pesos(resumen["diferencia_total"])],
        ["Facturas con diferencia", resumen["facturas_con_diferencia"]],
        ["Cobrado de menos (pérdida)", pesos(resumen["perdida"])],
        ["Cobrado de más", pesos(resumen["cobro_de_mas"])],
    ]
    mayores = [[f.factura_id, f.linea, f.plan_id, pesos(f.facturado), pesos(f.correcto), pesos(f.diferencia)]
               for f in diferencias.head(10).itertuples()]
    texto = "\n\n".join([
        f"# Informe de aseguramiento de ingresos · periodo {resumen['periodo']}",
        "## Resumen",
        tabla(["Indicador", "Valor"], indicadores),
        "## Mayores diferencias",
        tabla(["Factura", "Línea", "Plan", "Facturado", "Correcto", "Diferencia"], mayores)
        if mayores else "No se encontraron diferencias.",
        "![Diferencias por plan](diferencias.png)",
    ])
    (salida / "informe.md").write_text(texto + "\n", encoding="utf-8")
    return resumen


if __name__ == "__main__":
    print(generar_informe(Path(sys.argv[1]), Path(sys.argv[2])))
```

## Explicación

Cada archivo tiene un lector distinto, y por eso un formato distinto:

| Archivo | Lector | Por qué ese formato |
|---|---|---|
| `resumen.json` | Otro programa (un tablero, la tarea que envía alertas) | Estructurado, tipos claros |
| `diferencias.csv` | El analista que investiga | Se abre en Excel, se filtra |
| `diferencias.png` | La reunión de gerencia | Se entiende en 5 segundos |
| `informe.md` | Cualquiera, en VS Code o GitHub | Legible como texto, con tablas y la imagen |

Separar **pérdida** y **cobro de más** vuelve a aparecer (como en el ejercicio 4): una diferencia neta pequeña puede esconder dos problemas grandes de signo contrario. Y ordenar las diferencias por valor absoluto pone arriba lo que más importa, sea del signo que sea.

`matplotlib.use("Agg")` antes de importar `pyplot` es imprescindible en procesos automáticos: sin pantalla (una tarea programada, un contenedor de Docker), matplotlib intentaría abrir una ventana y fallaría.
