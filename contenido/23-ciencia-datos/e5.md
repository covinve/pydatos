## Enunciado

1. `prueba_permutacion(a, b, estadistico=np.mean, permutaciones=5000, semilla=0)` → diccionario `{"diferencia": d, "p": p}`:
   - `d = estadistico(a) − estadistico(b)`;
   - `rng = np.random.default_rng(semilla)`; `todas = np.concatenate([a, b])`;
   - en cada permutación: `mezcla = rng.permutation(todas)`, diferencia entre `estadistico(mezcla[:len(a)])` y `estadistico(mezcla[len(a):])`;
   - `p = (cantidad de diferencias permutadas con |valor| >= |d| + 1) / (permutaciones + 1)`.
2. `comparar_segmentos(cdr, clientes, estadistico=np.median)` → el resultado de la prueba comparando la duración (`duracion_seg > 0`) de las llamadas `VOZ` de clientes `EMPRESARIAL` (a) contra `RESIDENCIAL` (b).
3. `tamano_efecto(a, b)` → diferencia de medianas dividida por la `sigma_robusta` (`1.4826 × MAD`) de las dos muestras juntas, redondeada a 3 decimales.

## Pruebas

```python
import numpy as np
import pandas as pd

CDR = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"})
CLI = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})


def referencia(a, b, f, n, semilla):
    rng = np.random.default_rng(semilla)
    d = f(a) - f(b)
    todas = np.concatenate([a, b])
    c = sum(abs(f(m[:len(a)]) - f(m[len(a):])) >= abs(d) for m in (rng.permutation(todas) for _ in range(n)))
    return d, (c + 1) / (n + 1)


def prueba_algoritmo():
    """Reproduce el algoritmo exacto"""
    requiere("prueba_permutacion")
    rng = np.random.default_rng(1)
    a, b = rng.gamma(2, 60, 80), rng.gamma(2, 70, 120)
    r = prueba_permutacion(a, b, permutaciones=800, semilla=3)
    d, p = referencia(a, b, np.mean, 800, 3)
    cerca(r["diferencia"], d, 1e-9)
    cerca(r["p"], p, 1e-12)


def prueba_extremos():
    """Diferencia enorme → p mínimo; grupos iguales → p alto"""
    a = np.arange(100, 130, dtype=float)
    b = np.arange(0, 30, dtype=float)
    igual(prueba_permutacion(a, b, permutaciones=999)["p"], 1 / 1000)
    rng = np.random.default_rng(9)
    x = rng.normal(0, 1, 400)
    assert prueba_permutacion(x[:200], x[200:], permutaciones=999)["p"] > 0.05


def prueba_segmentos():
    """Empresarial contra residencial"""
    requiere("comparar_segmentos")
    v = CDR[(CDR["tipo"] == "VOZ") & (CDR["duracion_seg"] > 0)].merge(CLI, left_on="linea_origen", right_on="linea")
    a = v.loc[v["segmento"] == "EMPRESARIAL", "duracion_seg"].to_numpy()
    b = v.loc[v["segmento"] == "RESIDENCIAL", "duracion_seg"].to_numpy()
    r = comparar_segmentos(CDR, CLI)
    d, p = referencia(a, b, np.median, 5000, 0)
    cerca(r["diferencia"], d, 1e-9)
    cerca(r["p"], p, 1e-12)


def prueba_efecto():
    """Tamaño del efecto robusto"""
    requiere("tamano_efecto")
    a, b = np.array([10.0, 12, 14, 16, 18]), np.array([1.0, 2, 3, 4, 5])
    juntas = np.concatenate([a, b])
    sigma = 1.4826 * np.median(np.abs(juntas - np.median(juntas)))
    igual(tamano_efecto(a, b), round((14 - 3) / sigma, 3))
```

Para que tu resultado coincida **exactamente**, sigue el orden del enunciado: una sola llamada a `rng.permutation` por permutación, y la primera porción (`mezcla[:len(a)]`) es el grupo `a`. Devuelve `float(...)` en los dos valores.

## Solución

```python
import numpy as np
import pandas as pd


def prueba_permutacion(a, b, estadistico=np.mean, permutaciones=5000, semilla=0):
    a, b = np.asarray(a, dtype=float), np.asarray(b, dtype=float)
    rng = np.random.default_rng(semilla)
    observada = estadistico(a) - estadistico(b)
    todas = np.concatenate([a, b])
    extremas = 0
    for _ in range(permutaciones):
        mezcla = rng.permutation(todas)
        diferencia = estadistico(mezcla[:len(a)]) - estadistico(mezcla[len(a):])
        extremas += abs(diferencia) >= abs(observada)
    return {"diferencia": float(observada), "p": float((extremas + 1) / (permutaciones + 1))}


def _duraciones(cdr, clientes, segmento):
    voz = cdr[(cdr["tipo"] == "VOZ") & (cdr["duracion_seg"] > 0)]
    unidos = voz.merge(clientes, left_on="linea_origen", right_on="linea")
    return unidos.loc[unidos["segmento"] == segmento, "duracion_seg"].to_numpy()


def comparar_segmentos(cdr, clientes, estadistico=np.median):
    return prueba_permutacion(_duraciones(cdr, clientes, "EMPRESARIAL"), _duraciones(cdr, clientes, "RESIDENCIAL"),
                              estadistico=estadistico)


def tamano_efecto(a, b):
    a, b = np.asarray(a, dtype=float), np.asarray(b, dtype=float)
    juntas = np.concatenate([a, b])
    sigma = 1.4826 * np.median(np.abs(juntas - np.median(juntas)))
    return round(float((np.median(a) - np.median(b)) / sigma), 3)


cdr = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"})
clientes = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})
print(comparar_segmentos(cdr, clientes))
```

## Explicación

La prueba de permutación responde "¿podría esta diferencia ser azar?" **sin suponer nada** sobre la forma de los datos: ni normalidad ni varianzas iguales. Por eso funciona bien con duraciones de llamadas, con su cola larga, donde la clásica prueba t puede engañar. Y sirve con **cualquier** estadístico: comparamos medianas, que con datos sesgados describen mejor a los grupos.

El `+ 1` en el numerador y el denominador evita reportar `p = 0`: con 5.000 permutaciones, lo más que puedes afirmar es que `p < 1/5001`, no que la probabilidad sea cero.

`tamano_efecto` completa el análisis: un valor p dice si la diferencia es **creíble**; el tamaño del efecto dice si es **grande**. Como regla práctica, efectos menores a 0,2 desviaciones suelen ser irrelevantes para el negocio aunque sean "significativos", y por encima de 0,8 se consideran grandes. Ambos números juntos permiten una conclusión honesta.
