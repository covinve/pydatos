## Enunciado

`ejercicios/m19-e7/monolito.py` es un script de facturación real: lee, valida, calcula y escribe, todo en una función de 90 líneas. **Funciona** (sus totales cuadran con `esperado.json`), y por eso es la referencia: tu versión en capas debe producir **exactamente el mismo archivo, byte por byte**.

Reorganízalo en el paquete `ejercicios/m19-e7/facturacion/`. Cada archivo ya tiene un docstring con su contrato:

| Archivo | Capa | Puede importar |
|---|---|---|
| `dominio.py` | Dominio | Solo biblioteca estándar "pura" (`dataclasses`, `datetime`, `math`, `typing`…) |
| `puertos.py` | Aplicación | Lo anterior + `dominio` |
| `servicios.py` | Aplicación | Lo anterior + `puertos` |
| `adaptadores_csv.py` | Infraestructura | Cualquier cosa, menos `servicios` y `cli` |
| `cli.py` | Interfaz | Todo |

`test_capas.py` verifica el comportamiento **y** la arquitectura (lee los `import` de cada archivo con `ast`).

:::consejo Refactorizar con red
Trabaja en pasos pequeños y ejecuta las pruebas en cada uno: primero copia la lógica de cálculo a `calcular_factura` y prueba; luego la validación; luego los adaptadores. Nunca reescribas todo de una vez: así, cuando algo se rompe, sabes qué fue.
:::

## Pista 1

## Solución

```python facturacion/dominio.py
"""Dominio: las reglas de facturación. Sin archivos, sin prints, sin csv ni pandas."""

from __future__ import annotations

import math
from collections.abc import Iterable
from dataclasses import dataclass
from datetime import date, datetime

TIPOS = ("VOZ", "SMS", "DATOS")
FORMATO_FECHA = "%Y-%m-%d %H:%M:%S"
CORTE_ANTIGUEDAD = date(2023, 9, 1)
ILIMITADO = -1


@dataclass(frozen=True)
class Plan:
    plan_id: str
    cargo_fijo: int
    minutos_incluidos: int
    mb_incluidos: int
    sms_incluidos: int
    tarifa_minuto: int
    tarifa_minuto_internacional: int
    tarifa_mb: int
    tarifa_sms: int


@dataclass(frozen=True)
class Cliente:
    linea: str
    plan_id: str
    segmento: str
    fecha_alta: date


@dataclass(frozen=True)
class Consumo:
    minutos_nacionales: int = 0
    minutos_internacionales: int = 0
    mb: float = 0.0
    sms: int = 0


@dataclass(frozen=True)
class Factura:
    linea: str
    plan_id: str
    cargo_fijo: int
    valor_voz: int
    valor_internacional: int
    valor_datos: int
    valor_sms: int
    descuento: int
    subtotal: int
    iva: int
    total: int


def minutos_facturables(segundos: int) -> int:
    return (segundos + 59) // 60


def redondear(valor: float) -> int:
    return int(valor + 0.5)


def cdr_valido(cdr: dict, lineas: set[str], vistos: set[str]) -> bool:
    if cdr["cdr_id"] in vistos:
        return False
    vistos.add(cdr["cdr_id"])
    if not cdr["linea_origen"] or cdr["linea_origen"] not in lineas:
        return False
    if cdr["tipo"] not in TIPOS or int(cdr["duracion_seg"]) < 0:
        return False
    try:
        datetime.strptime(cdr["fecha_inicio"], FORMATO_FECHA)
    except ValueError:
        return False
    return True


def consumo_por_linea(cdrs: Iterable[dict], lineas: set[str]) -> dict[str, Consumo]:
    acumulado: dict[str, dict] = {}
    vistos: set[str] = set()
    for cdr in cdrs:
        if not cdr_valido(cdr, lineas, vistos):
            continue
        c = acumulado.setdefault(cdr["linea_origen"], {"nac": 0, "int": 0, "mb": 0.0, "sms": 0})
        if cdr["tipo"] == "VOZ":
            clave = "int" if cdr["destino"] == "INTERNACIONAL" else "nac"
            c[clave] += minutos_facturables(int(cdr["duracion_seg"]))
        elif cdr["tipo"] == "DATOS":
            c["mb"] += float(cdr["mb"])
        else:
            c["sms"] += 1
    return {linea: Consumo(c["nac"], c["int"], c["mb"], c["sms"]) for linea, c in acumulado.items()}


def _excedente(usado: int, incluido: int, tarifa: int) -> int:
    return 0 if incluido == ILIMITADO else max(0, usado - incluido) * tarifa


def calcular_factura(cliente: Cliente, plan: Plan, consumo: Consumo) -> Factura:
    voz = _excedente(consumo.minutos_nacionales, plan.minutos_incluidos, plan.tarifa_minuto)
    internacional = consumo.minutos_internacionales * plan.tarifa_minuto_internacional
    datos = _excedente(math.ceil(round(consumo.mb, 6)), plan.mb_incluidos, plan.tarifa_mb)
    sms = _excedente(consumo.sms, plan.sms_incluidos, plan.tarifa_sms)
    descuento = (redondear(plan.cargo_fijo * 0.10)
                 if cliente.segmento == "EMPRESARIAL" and cliente.fecha_alta < CORTE_ANTIGUEDAD else 0)
    subtotal = plan.cargo_fijo + voz + internacional + datos + sms - descuento
    iva = redondear(subtotal * 0.19)
    return Factura(cliente.linea, plan.plan_id, plan.cargo_fijo, voz, internacional, datos, sms,
                   descuento, subtotal, iva, subtotal + iva)
```

```python facturacion/puertos.py
"""Puertos: lo que el servicio necesita del exterior, como Protocol."""

from __future__ import annotations

from collections.abc import Iterable
from typing import Protocol

from facturacion.dominio import Cliente, Factura, Plan


class FuenteDatos(Protocol):
    def planes(self) -> dict[str, Plan]: ...
    def clientes(self) -> list[Cliente]: ...
    def cdrs(self) -> Iterable[dict]: ...


class SalidaFacturas(Protocol):
    def guardar(self, facturas: list[Factura]) -> None: ...
```

```python facturacion/servicios.py
"""Caso de uso: facturar el periodo."""

from __future__ import annotations

from facturacion.dominio import Consumo, calcular_factura, consumo_por_linea
from facturacion.puertos import FuenteDatos, SalidaFacturas


def facturar(fuente: FuenteDatos, salida: SalidaFacturas) -> dict:
    planes = fuente.planes()
    clientes = fuente.clientes()
    consumos = consumo_por_linea(fuente.cdrs(), {c.linea for c in clientes})
    facturas = [calcular_factura(c, planes[c.plan_id], consumos.get(c.linea, Consumo())) for c in clientes]
    salida.guardar(facturas)
    return {"facturas": len(facturas), "total": sum(f.total for f in facturas)}
```

```python facturacion/adaptadores_csv.py
"""Adaptadores CSV."""

from __future__ import annotations

import csv
from collections.abc import Iterator
from dataclasses import astuple, fields
from datetime import date
from pathlib import Path

from facturacion.dominio import Cliente, Factura, Plan


class FuenteCSV:
    def __init__(self, carpeta):
        self.carpeta = Path(carpeta)

    def planes(self) -> dict[str, Plan]:
        with open(self.carpeta / "planes.csv", encoding="utf-8", newline="") as f:
            return {
                p["plan_id"]: Plan(p["plan_id"], int(p["cargo_fijo"]), int(p["minutos_incluidos"]),
                                   int(p["mb_incluidos"]), int(p["sms_incluidos"]), int(p["tarifa_minuto"]),
                                   int(p["tarifa_minuto_internacional"]), int(p["tarifa_mb"]), int(p["tarifa_sms"]))
                for p in csv.DictReader(f)
            }

    def clientes(self) -> list[Cliente]:
        with open(self.carpeta / "clientes.csv", encoding="latin-1", newline="") as f:
            return [Cliente(c["linea"], c["plan_id"], c["segmento"], date.fromisoformat(c["fecha_alta"]))
                    for c in csv.DictReader(f)]

    def cdrs(self) -> Iterator[dict]:
        with open(self.carpeta / "cdr.csv", encoding="utf-8", newline="") as f:
            yield from csv.DictReader(f, delimiter="|")


class SalidaCSV:
    def __init__(self, ruta):
        self.ruta = Path(ruta)

    def guardar(self, facturas: list[Factura]) -> None:
        self.ruta.parent.mkdir(parents=True, exist_ok=True)
        with open(self.ruta, "w", encoding="utf-8", newline="") as f:
            escritor = csv.writer(f)
            escritor.writerow([campo.name for campo in fields(Factura)])
            escritor.writerows(astuple(factura) for factura in facturas)
```

```python facturacion/cli.py
"""Raíz de composición.

Uso:
    python -m facturacion.cli datos_prueba salidas/facturas.csv     (desde ejercicios/m19-e7)
"""

from __future__ import annotations

import sys

from facturacion.adaptadores_csv import FuenteCSV, SalidaCSV
from facturacion.servicios import facturar


def main(argv: list[str] | None = None) -> int:
    carpeta, ruta_salida = argv if argv is not None else sys.argv[1:3]
    resumen = facturar(FuenteCSV(carpeta), SalidaCSV(ruta_salida))
    total = f"{resumen['total']:,}".replace(",", ".")
    print(f"{resumen['facturas']} facturas, total ${total}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## Explicación

Mira lo que ganaste, sin cambiar **un solo peso** del resultado:

- **`calcular_factura` se prueba con una línea**: la prueba del ejemplo de la referencia (342 minutos en HOG30) no necesita ningún archivo.
- **`facturar` se prueba con dobles** (`FuenteMemoria`, `SalidaMemoria`): el caso de uso completo, en milisegundos.
- **Cambiar de fuente es escribir un adaptador**: `FuenteSQLite` o `FuenteDuckDB` (para los 20 GB) implementarían los mismos tres métodos; dominio y servicio no se tocan.
- **Las reglas están en un lugar**: `CORTE_ANTIGUEDAD`, `ILIMITADO`, `_excedente`. El monolito repetía la lógica de "ilimitado" tres veces.

La prueba de "mismo archivo byte por byte" es la red de seguridad de toda refactorización: un **test de caracterización** que captura lo que el sistema hace hoy, para asegurar que reorganizarlo no lo cambia. Es la forma más segura de modernizar código heredado que nadie quiere tocar.

Y las pruebas de arquitectura evitan que el diseño se degrade con el tiempo: el día que alguien escriba `import csv` en el dominio "solo por esta vez", la prueba falla y obliga a la conversación.
