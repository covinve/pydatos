import numpy as np
import pandas as pd

CDR = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"})
CLI = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})


def prueba_z_robusto():
    """La fórmula y el caso MAD = 0"""
    requiere("z_robusto")
    s = pd.Series([10.0, 12, 11, 13, 100])
    mediana, mad = 12.0, 1.0
    igual([round(v, 4) for v in z_robusto(s)], [round(0.6745 * (v - mediana) / mad, 4) for v in s])
    igual(list(z_robusto(pd.Series([5.0, 5, 5]))), [0.0, 0.0, 0.0])
    igual(list(z_robusto(pd.Series([1.0, 2, 3], index=["a", "b", "c"])).index), ["a", "b", "c"], "Conserva el índice")


def prueba_consumo():
    """Consumo de datos por línea cliente"""
    requiere("consumo_datos")
    c = consumo_datos(CDR, CLI)
    igual(list(c.columns), ["linea", "plan_id", "mb"])
    d = CDR[(CDR["tipo"] == "DATOS") & (CDR["duracion_seg"] >= 0)].merge(CLI, left_on="linea_origen", right_on="linea")
    e = d.groupby("linea")["mb"].sum()
    igual(len(c), len(e))
    igual(c["linea"].tolist(), sorted(e.index))
    cerca(float(c["mb"].sum()), float(e.sum()), 0.01)


def prueba_anomalos_plantados():
    """Encuentra las anomalías plantadas en su plan"""
    requiere("anomalos")
    c = consumo_datos(CDR, CLI).copy()
    planes = c["plan_id"].value_counts()
    plan = planes.index[0]
    fila = c.index[c["plan_id"] == plan][0]

    mediana = serie.median()

    if mad == 0:
        return serie * 0.0

## Solución

```python
import numpy as np
import pandas as pd


def z_robusto(serie):
    mediana = serie.median()
    mad = (serie - mediana).abs().median()
    if mad == 0:
        return serie * 0.0
    return 0.6745 * (serie - mediana) / mad


def consumo_datos(cdr, clientes):
    datos = cdr[(cdr["tipo"] == "DATOS") & (cdr["duracion_seg"] >= 0)]
    unidos = datos.merge(clientes[["linea", "plan_id"]], left_on="linea_origen", right_on="linea")
    return (unidos.groupby(["linea", "plan_id"], as_index=False)["mb"].sum()
            .sort_values("linea").reset_index(drop=True))


def anomalos(consumo, umbral=3.5):
    z = consumo.groupby("plan_id")["mb"].transform(z_robusto)
    r = consumo.assign(z=z.round(2))
    return r[z.abs() > umbral].sort_values("z", ascending=False).reset_index(drop=True)


cdr = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"})
clientes = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})
consumo = consumo_datos(cdr, clientes)
print(anomalos(consumo, umbral=2))
```

## Explicación

La última prueba resume la idea del ejercicio: 2.000 MB es una anomalía en un plan básico donde todos consumen unos 100, y no lo sería si se comparara con el plan ilimitado. Calcular el z de toda la base mezclada escondería al cliente básico entre los ilimitados, y marcaría como "raros" a todos los ilimitados.

La prueba de anomalías **plantadas** es la forma honesta de probar un detector: no sabes qué es anómalo en datos reales, pero sí puedes inyectar casos que **deben** encontrarse y verificar que aparecen. La usarás de nuevo con Isolation Forest en el ejercicio 7.

Que `MAD = 0` devuelva ceros no es un detalle: en un plan con 3 líneas que consumen exactamente lo mismo, dividir por cero daría infinitos o `NaN`, y la línea siguiente del pipeline fallaría (o peor, marcaría a todas como anómalas).
