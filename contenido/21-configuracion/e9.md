## Enunciado

Junta el módulo en una herramienta: una CLI de Typer cuya configuración llega por **cinco capas** con la precedencia de la lección 3, validada con pydantic-settings, con un comando para **ver** la configuración efectiva (sin secretos) y otro para **validarla**.

El docstring de `ejercicios/m21-e9/telco_config.py` describe:

1. la clase `Config` (campos, reglas y precedencia);
2. `crear_config(toml, env_file, **argumentos)`;
3. los comandos `mostrar` (con `--json`), `validar` y `ejecutar`, con sus salidas y códigos.

:::consejo Un archivo TOML elegido al ejecutar
`TomlConfigSettingsSource` lee la ruta de `model_config["toml_file"]`, que es fija en la clase. Para elegir el archivo al ejecutar, crea una **subclase** al vuelo con esa ruta:

```python
clase = type("ConfigConArchivo", (Config,), {"model_config": {**Config.model_config, "toml_file": toml}})
return clase(_env_file=env_file, **argumentos)
```
:::

## Pista 1

El orden de las fuentes es el orden de precedencia (la primera gana):

```python
@classmethod
def settings_customise_sources(cls, settings_cls, init_settings, env_settings, dotenv_settings,
                               file_secret_settings):
    return (init_settings, env_settings, dotenv_settings, TomlConfigSettingsSource(settings_cls))
```

## Pista 2

Para los errores, `ValidationError.errors()` da una lista de diccionarios con `loc` (el campo) y `msg`:

```python
def _cargar_o_salir(toml, env_file, **argumentos):
    try:
        return crear_config(toml, env_file, **argumentos)
    except ValidationError as error:
        for e in error.errors():
            typer.echo(f"✗ {e['loc'][0]}: {e['msg']}")
        raise typer.Exit(code=2)
```

Para mostrar en el orden de la clase: `for campo in Config.model_fields:`.

## Solución

```python telco_config.py
"""Ejercicio m21-e9: una CLI configurable por capas (TOML < .env < entorno < argumentos).

Uso:
    python ejercicios/m21-e9/telco_config.py mostrar --config config.toml
    python ejercicios/m21-e9/telco_config.py validar --config config.toml --env-file .env
    python ejercicios/m21-e9/telco_config.py ejecutar --config config.toml --umbral 1.5
"""

from __future__ import annotations

import json
import re
from pathlib import Path
from typing import Annotated, Optional

import typer
from pydantic import Field, SecretStr, ValidationError, field_validator
from pydantic_settings import BaseSettings, SettingsConfigDict, TomlConfigSettingsSource


class Config(BaseSettings):
    model_config = SettingsConfigDict(env_prefix="TELCO_", extra="forbid", frozen=True)

    periodo: str = "202609"
    datos: Path = Path("datos")
    umbral_rechazo: float = Field(5.0, ge=0, le=100)
    hilos: int = Field(4, ge=1, le=64)
    db_url: SecretStr | None = None

    @field_validator("periodo")
    @classmethod
    def _periodo(cls, valor: str) -> str:
        if not re.fullmatch(r"\d{4}(0[1-9]|1[0-2])", valor):
            raise ValueError("el periodo debe tener formato AAAAMM (mes 01 a 12)")
        return valor

    @classmethod
    def settings_customise_sources(cls, settings_cls, init_settings, env_settings, dotenv_settings,
                                   file_secret_settings):
        return (init_settings, env_settings, dotenv_settings, TomlConfigSettingsSource(settings_cls))

    def para_mostrar(self) -> dict[str, object]:
        valores = {campo: getattr(self, campo) for campo in type(self).model_fields}
        valores["db_url"] = "***" if self.db_url else ""
        return valores


def crear_config(toml: Path | None, env_file: Path | None, **argumentos: object) -> Config:
    clase = type("ConfigConArchivo", (Config,), {"model_config": {**Config.model_config, "toml_file": toml}})
    return clase(_env_file=env_file, **{k: v for k, v in argumentos.items() if v is not None})


app = typer.Typer(help="Configuración del pipeline de facturación.", no_args_is_help=True)

OpcionConfig = Annotated[Optional[Path], typer.Option("--config", help="Archivo TOML", exists=True, dir_okay=False)]
OpcionEnv = Annotated[Optional[Path], typer.Option("--env-file", help="Archivo .env", exists=True, dir_okay=False)]


def _cargar_o_salir(toml: Path | None, env_file: Path | None, **argumentos: object) -> Config:
    try:
        return crear_config(toml, env_file, **argumentos)
    except ValidationError as error:
        for e in error.errors():
            typer.echo(f"✗ {e['loc'][0]}: {e['msg']}")
        raise typer.Exit(code=2) from None


@app.command()
def mostrar(config: OpcionConfig = None, env_file: OpcionEnv = None,
            como_json: Annotated[bool, typer.Option("--json", help="Salida en JSON")] = False):
    """Muestra la configuración efectiva (sin secretos)."""
    valores = _cargar_o_salir(config, env_file).para_mostrar()
    if como_json:
        typer.echo(json.dumps(valores, default=str, ensure_ascii=False, indent=2))
    else:
        for campo, valor in valores.items():
            typer.echo(f"{campo} = {valor}")


@app.command()
def validar(config: OpcionConfig = None, env_file: OpcionEnv = None):
    """Valida la configuración y explica los errores."""
    _cargar_o_salir(config, env_file)
    typer.echo("Configuración válida")


@app.command()
def ejecutar(config: OpcionConfig = None, env_file: OpcionEnv = None,
             periodo: Annotated[Optional[str], typer.Option(help="Periodo AAAAMM")] = None,
             umbral: Annotated[Optional[float], typer.Option(help="Umbral de rechazo (%)")] = None):
    """Ejecuta el proceso (simulado) con la configuración efectiva."""
    c = _cargar_o_salir(config, env_file, periodo=periodo, umbral_rechazo=umbral)
    typer.echo(f"Procesando {c.datos} periodo {c.periodo} con umbral {c.umbral_rechazo} y {c.hilos} hilos")


if __name__ == "__main__":
    app()
```

## Explicación

La herramienta tiene ahora el comportamiento que esperarías de un programa profesional:

```powershell
python telco_config.py mostrar --config config.toml --env-file .env
periodo = 202608
datos = D:\cdr
umbral_rechazo = 3.0
hilos = 12            ← ¿12? El TOML dice 8… → hay un TELCO_HILOS=12 en el .env
db_url = ***
```

`mostrar` es el comando que más vas a agradecer: ante "¿con qué configuración corrió?", la respuesta está a un comando, sin exponer la contraseña. `validar` sirve como primer paso de una tarea programada o de un *pipeline* de CI: si alguien rompió el `config.toml`, falla en un segundo con el campo exacto.

`extra="forbid"` hace que una clave mal escrita en el TOML (`umbrall`) sea un error en lugar de ignorarse en silencio, lo mismo que hiciste a mano en el ejercicio 2.

Y fíjate en la división de responsabilidades: `Config` sabe validar, `crear_config` sabe combinar fuentes, y los comandos solo traducen opciones en argumentos y errores en mensajes. Es la raíz de composición del módulo 19 con configuración incluida: exactamente lo que necesitará la CLI del proyecto final.
