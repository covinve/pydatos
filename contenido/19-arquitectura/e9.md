## Enunciado

La prueba de fuego de una arquitectura: **agregar una funcionalidad nueva sin modificar la que ya existe**.

`ejercicios/m19-e9/facturacion/` contiene la facturación en capas del ejercicio 7, terminada. Ahora el negocio pide la **conciliación** (lo que hiciste en el proyecto 2 con Pandas): comparar las facturas emitidas con las correctas y reportar las diferencias.

Escribe tres archivos **nuevos** (cada uno trae su contrato en el docstring):

| Archivo | Capa | Contenido |
|---|---|---|
| `dominio_conciliacion.py` | Dominio | `FacturaEmitida`, `Hallazgo`, `revisar`, `resumir` |
| `conciliar.py` | Aplicación | Los puertos `FacturasEmitidas` y `SalidaConciliacion`, y el caso de uso `conciliar` |
| `adaptadores_conciliacion.py` | Infraestructura | `EmitidasCSV` y `ReporteJSON` |

El resumen con los datos de prueba debe ser **idéntico** a la sección `"facturacion"` de `esperado.json`. Y no puedes tocar `dominio.py`, `servicios.py` ni los adaptadores de facturación: los **reutilizas**.

## Pista 1

En `conciliar`, las facturas correctas salen del dominio existente, sin pasar por `servicios.facturar` (que exige una salida):


## Pista 2

En `revisar`, tres comparaciones independientes:

```python
consumo_emitido = e.valor_voz + e.valor_internacional + e.valor_datos + e.valor_sms
consumo_correcto = c.valor_voz + c.valor_internacional + c.valor_datos + c.valor_sms
if consumo_emitido != consumo_correcto:
    hallazgos.append(Hallazgo(e.factura_id, e.linea, "consumo", consumo_emitido, consumo_correcto))
if e.total != e.subtotal + e.iva:
    ...
```

`resumir` se arma recorriendo los pares y usando `revisar` para saber de qué tipo es cada uno. Para el JSON: `dataclasses.asdict(hallazgo)`.

## Solución

```python facturacion/dominio_conciliacion.py
"""Dominio de la conciliación: comparar lo que se facturó con lo que se debió facturar."""

from __future__ import annotations

from dataclasses import dataclass

from facturacion.dominio import Factura


@dataclass(frozen=True)
class FacturaEmitida:
    factura_id: str
    linea: str
    valor_voz: int
    valor_internacional: int
    valor_datos: int
    valor_sms: int
    descuento: int
    subtotal: int
    iva: int
    total: int


@dataclass(frozen=True)
class Hallazgo:
    factura_id: str
    linea: str
    tipo: str
    emitido: int
    correcto: int


def _consumo(f) -> int:
    return f.valor_voz + f.valor_internacional + f.valor_datos + f.valor_sms


def revisar(emitida: FacturaEmitida, correcta: Factura) -> list[Hallazgo]:
    def hallazgo(tipo, emitido, correcto):
        return Hallazgo(emitida.factura_id, emitida.linea, tipo, emitido, correcto)

    hallazgos = []
    if _consumo(emitida) != _consumo(correcta):
        hallazgos.append(hallazgo("consumo", _consumo(emitida), _consumo(correcta)))
    if emitida.total != emitida.subtotal + emitida.iva:
        hallazgos.append(hallazgo("inconsistente", emitida.total, emitida.subtotal + emitida.iva))
    if emitida.total != correcta.total:
        hallazgos.append(hallazgo("total", emitida.total, correcta.total))
    return hallazgos


def resumir(pares: list[tuple[FacturaEmitida, Factura]]) -> dict:
    r = {"facturas": 0, "total_facturado": 0, "total_correcto": 0, "facturas_con_diferencia_consumo": 0,
         "facturas_inconsistentes": 0, "facturas_total_distinto": 0, "diferencia_total": 0,
         "diferencia_por_consumo": 0, "iva_correcto": 0}
    for emitida, correcta in pares:
        tipos = {h.tipo for h in revisar(emitida, correcta)}
        diferencia = emitida.total - correcta.total
        r["facturas"] += 1
        r["total_facturado"] += emitida.total
        r["total_correcto"] += correcta.total
        r["iva_correcto"] += correcta.iva
        r["diferencia_total"] += diferencia
        r["facturas_con_diferencia_consumo"] += "consumo" in tipos
        r["facturas_inconsistentes"] += "inconsistente" in tipos
        r["facturas_total_distinto"] += "total" in tipos
        if "consumo" in tipos:
            r["diferencia_por_consumo"] += diferencia
    return r
```

```python facturacion/conciliar.py
"""Caso de uso: conciliar el periodo."""

from __future__ import annotations

from typing import Protocol

from facturacion.dominio import Consumo, calcular_factura, consumo_por_linea
from facturacion.dominio_conciliacion import FacturaEmitida, Hallazgo, resumir, revisar
from facturacion.puertos import FuenteDatos


class FacturasEmitidas(Protocol):
    def todas(self) -> list[FacturaEmitida]: ...


class SalidaConciliacion(Protocol):
    def publicar(self, resumen: dict, hallazgos: list[Hallazgo]) -> None: ...


def conciliar(fuente: FuenteDatos, emitidas: FacturasEmitidas, salida: SalidaConciliacion) -> dict:
    planes = fuente.planes()
    clientes = fuente.clientes()
    consumos = consumo_por_linea(fuente.cdrs(), {c.linea for c in clientes})
    correctas = {c.linea: calcular_factura(c, planes[c.plan_id], consumos.get(c.linea, Consumo())) for c in clientes}
    pares = [(e, correctas[e.linea]) for e in emitidas.todas()]
    resumen = resumir(pares)
    salida.publicar(resumen, [h for e, c in pares for h in revisar(e, c)])
    return resumen
```

```python facturacion/adaptadores_conciliacion.py
"""Adaptadores de la conciliación."""

from __future__ import annotations

import csv
import json
from dataclasses import asdict
from pathlib import Path

from facturacion.dominio_conciliacion import FacturaEmitida, Hallazgo

ENTEROS = ["valor_voz", "valor_internacional", "valor_datos", "valor_sms", "descuento", "subtotal", "iva", "total"]


class EmitidasCSV:
    def __init__(self, ruta):
        self.ruta = Path(ruta)

    def todas(self) -> list[FacturaEmitida]:
        with open(self.ruta, encoding="utf-8", newline="") as f:
            return [FacturaEmitida(fila["factura_id"], fila["linea"], *(int(fila[c]) for c in ENTEROS))
                    for fila in csv.DictReader(f)]


class ReporteJSON:
    def __init__(self, ruta):
        self.ruta = Path(ruta)

    def publicar(self, resumen: dict, hallazgos: list[Hallazgo]) -> None:
        self.ruta.parent.mkdir(parents=True, exist_ok=True)
        contenido = {"resumen": resumen, "hallazgos": [asdict(h) for h in hallazgos]}
        self.ruta.write_text(json.dumps(contenido, ensure_ascii=False, indent=2), encoding="utf-8")
```

## Explicación

Lo más importante de este ejercicio es lo que **no** hiciste: no tocaste una sola línea de la facturación. La conciliación se apoyó en `calcular_factura`, `consumo_por_linea` y `FuenteCSV`, que ya estaban probados. Eso es abierto/cerrado a escala de sistema.

Compara con la versión del proyecto 2: allí la conciliación y la facturación vivían en el mismo flujo de Pandas, y para probar la regla de "inconsistente" necesitabas un CSV con una factura mala. Aquí `revisar` se prueba con dos objetos escritos a mano, y `conciliar` con tres dobles.

¿Y el rendimiento con millones de facturas? El diseño no lo impide: un `FuenteDuckDB` podría calcular el consumo en SQL y un `EmitidasParquet` leer por lotes. Pero ahí aparece una decisión de arquitectura real: si el consumo se calcula en SQL, la regla de validación queda **duplicada** (en SQL y en Python). Las opciones son aceptar la duplicación y vigilarla con una prueba que compare ambas implementaciones (como las pruebas de contrato del ejercicio 8), o mover la regla a un solo lugar. No hay respuesta universal: lo importante es tomar la decisión a conciencia. En el proyecto final te tocará tomarla.
