## Enunciado

El área de aseguramiento de ingresos sospecha que algunas llamadas internacionales no se están cobrando. Construye el control.

1. `minutos_internacionales(cdr, clientes)` → Serie indexada por `linea` con los minutos facturables (`(duracion_seg + 59) // 60` por llamada) de los CDR **válidos** de `VOZ` con destino `INTERNACIONAL`. Válido = no duplicado (primera aparición del `cdr_id`), línea de un cliente, `duracion_seg >= 0` y fecha válida.
2. `fugas_internacional(cdr, clientes, planes, facturas)` → DataFrame con una fila por **factura** cuyo `valor_internacional` difiere de lo esperado (`minutos × tarifa_minuto_internacional` del plan de la factura; 0 minutos si la línea no llamó al exterior). Columnas: `factura_id`, `linea`, `minutos`, `esperado`, `facturado`, `diferencia` (`esperado − facturado`, positiva = **se dejó de cobrar**). Ordenado por `diferencia` de mayor a menor.
3. `resumen_fugas(fugas)` → `{"facturas": n, "perdida": suma de diferencias positivas, "cobro_de_mas": suma de |diferencias negativas|}` (enteros).

import pandas as pd

CDR = pd.read_csv("cdr.csv", sep="|", dtype=str, keep_default_na=False)
CLI = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": str})
PLA = pd.read_csv("planes.csv")
FAC = pd.read_csv("facturas.csv", dtype={"linea": str})


def esperado_minutos():
    d = CDR[~CDR["cdr_id"].duplicated()]
    dur = pd.to_numeric(d["duracion_seg"])
    fecha = pd.to_datetime(d["fecha_inicio"], format="%Y-%m-%d %H:%M:%S", errors="coerce")
    ok = d["linea_origen"].isin(CLI["linea"]) & (dur >= 0) & fecha.notna() & (d["tipo"] == "VOZ") & (d["destino"] == "INTERNACIONAL")
    return ((dur[ok] + 59) // 60).groupby(d.loc[ok, "linea_origen"]).sum()


def prueba_minutos():
    """Minutos internacionales de CDR válidos"""
    requiere("minutos_internacionales")
    r = minutos_internacionales(CDR, CLI)
    e = esperado_minutos()
    igual(int(r.sum()), int(e.sum()))
    igual(sorted(r.index), sorted(e.index))
    linea = e.idxmax()
    igual(int(r[linea]), int(e[linea]))


def prueba_fuga_plantada():
    """Detecta una factura a la que se le borró el cobro internacional"""
    requiere("fugas_internacional", "resumen_fugas")
    e = esperado_minutos()
    linea = e.idxmax()
    facturas = FAC.copy()
    fila = facturas.index[facturas["linea"] == linea][0]
    tarifa = int(PLA.set_index("plan_id").loc[facturas.loc[fila, "plan_id"], "tarifa_minuto_internacional"])
    base = fugas_internacional(CDR, CLI, PLA, FAC)
    facturas.loc[fila, "valor_internacional"] = 0
    otra = facturas.index[facturas["linea"] != linea][0]
    facturas.loc[otra, "valor_internacional"] = facturas.loc[otra, "valor_internacional"] + 950
    r = fugas_internacional(CDR, CLI, PLA, facturas)
    igual(list(r.columns), ["factura_id", "linea", "minutos", "esperado", "facturado", "diferencia"])
    primera = r.iloc[0]
    igual((primera["linea"], int(primera["minutos"]), int(primera["esperado"]), int(primera["facturado"])),
          (linea, int(e[linea]), int(e[linea]) * tarifa, 0))
    igual(int(primera["diferencia"]), int(e[linea]) * tarifa)
    assert facturas.loc[otra, "factura_id"] in r["factura_id"].tolist(), "También detecta cobros de más"
    igual(r["diferencia"].tolist(), sorted(r["diferencia"].tolist(), reverse=True))
    s = resumen_fugas(r)
    sb = resumen_fugas(base)
    igual(s["facturas"], len(r))
    assert s["perdida"] >= sb["perdida"] + int(e[linea]) * tarifa - 950
    assert s["cobro_de_mas"] >= 950


def prueba_resumen_vacio():
    """Sin fugas"""
    vacio = pd.DataFrame(columns=["factura_id", "linea", "minutos", "esperado", "facturado", "diferencia"])
    igual(resumen_fugas(vacio), {"facturas": 0, "perdida": 0, "cobro_de_mas": 0})

## Solución

```python
import pandas as pd


def minutos_internacionales(cdr, clientes):
    d = cdr[~cdr["cdr_id"].duplicated()]
    duracion = pd.to_numeric(d["duracion_seg"])
    fecha = pd.to_datetime(d["fecha_inicio"], format="%Y-%m-%d %H:%M:%S", errors="coerce")
    ok = (d["linea_origen"].isin(clientes["linea"]) & (duracion >= 0) & fecha.notna()
          & (d["tipo"] == "VOZ") & (d["destino"] == "INTERNACIONAL"))
    minutos = (duracion[ok] + 59) // 60
    return minutos.groupby(d.loc[ok, "linea_origen"].rename("linea")).sum()


def fugas_internacional(cdr, clientes, planes, facturas):
    minutos = minutos_internacionales(cdr, clientes)
    f = facturas.merge(planes[["plan_id", "tarifa_minuto_internacional"]], on="plan_id")
    f["minutos"] = f["linea"].map(minutos).fillna(0).astype(int)
    f["esperado"] = f["minutos"] * f["tarifa_minuto_internacional"]
    f["facturado"] = f["valor_internacional"]
    f["diferencia"] = f["esperado"] - f["facturado"]
    columnas = ["factura_id", "linea", "minutos", "esperado", "facturado", "diferencia"]
    return f.loc[f["diferencia"] != 0, columnas].sort_values("diferencia", ascending=False).reset_index(drop=True)


def resumen_fugas(fugas):
    d = fugas["diferencia"].astype(int)
    return {"facturas": int(len(fugas)), "perdida": int(d[d > 0].sum()), "cobro_de_mas": int(-d[d < 0].sum())}


cdr = pd.read_csv("cdr.csv", sep="|", dtype=str, keep_default_na=False)
clientes = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": str})
planes = pd.read_csv("planes.csv")
facturas = pd.read_csv("facturas.csv", dtype={"linea": str})
f = fugas_internacional(cdr, clientes, planes, facturas)
print(f)
print(resumen_fugas(f))
```

## Explicación

Este control es la conciliación del proyecto 2 **enfocada** en un concepto y presentada como lo pide un área de aseguramiento de ingresos: no "hay diferencias", sino **cuánto dinero** se dejó de cobrar (y cuánto se cobró de más, que es un riesgo regulatorio y de reclamos).

Separar `perdida` y `cobro_de_mas` en lugar de sumar la diferencia neta es importante: 10 millones dejados de cobrar y 10 millones cobrados de más suman cero, pero son dos problemas distintos, con dos responsables distintos.

Ordenar por diferencia pone arriba los casos de mayor impacto: con miles de facturas, nadie revisa todo; se revisan primero las 20 que más dinero explican (en estos procesos, unas pocas facturas suelen concentrar la mayor parte de la pérdida).
