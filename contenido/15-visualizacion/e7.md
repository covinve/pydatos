## Enunciado

Arma el **tablero mensual de calidad de datos** en una sola figura de 2 × 2 y devuélvela junto con su imagen PNG. Escribe `tablero(cdr, clientes)` que devuelva `(fig, png)` donde `png` son los bytes de la imagen (`fig.savefig` sobre un `io.BytesIO`, `dpi=100`):

| Posición | Gráfico | Título exacto |
|---|---|---|
| `[0, 0]` | Barras horizontales: cantidad de registros por motivo de rechazo (las 6 reglas del curso; solo los motivos con al menos un registro), ordenadas de menor a mayor | `Registros rechazados por motivo` |
| `[0, 1]` | Línea: registros válidos por día | `Registros válidos por día` |
| `[1, 0]` | Barras: participación (%) de cada tipo en los registros válidos, en orden `VOZ`, `SMS`, `DATOS` | `Composición del tráfico válido (%)` |
| `[1, 1]` | Barras: GB de datos válidos por hora del día (24 barras, horas sin datos en 0) | `GB por hora del día` |

Título general: `Calidad de datos · septiembre 2026`.

import numpy as np
import pandas as pd

CDR = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"}, keep_default_na=False)
CLI = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})
F = pd.to_datetime(CDR["fecha_inicio"], format="%Y-%m-%d %H:%M:%S", errors="coerce")
COND = [CDR.duplicated("cdr_id"), CDR["linea_origen"] == "", ~CDR["linea_origen"].isin(CLI["linea"]),
        ~CDR["tipo"].isin(["VOZ", "SMS", "DATOS"]), CDR["duracion_seg"] < 0, F.isna()]
MOTIVO = pd.Series(np.select(COND, ["duplicado", "linea_vacia", "linea_desconocida", "tipo_desconocido",
                                    "duracion_negativa", "fecha_invalida"], default="valido"))
VAL = CDR[MOTIVO.to_numpy() == "valido"]


_TABLERO = []


def ejes():
    """Construye el tablero una sola vez y lo reutiliza en todas las pruebas."""
    if not _TABLERO:
        fig, png = tablero(CDR, CLI)
        _TABLERO.append((fig, png, fig.axes))
    return _TABLERO[0]


def prueba_estructura():
    """Cuatro gráficos con sus títulos y la imagen PNG"""
    fig, png, a = ejes()
    assert len(a) >= 4
    igual([x.get_title() for x in a[:4]], ["Registros rechazados por motivo", "Registros válidos por día",
                                          "Composición del tráfico válido (%)", "GB por hora del día"])
    assert png[:8] == b"\x89PNG\r\n\x1a\n", "png deben ser los bytes de una imagen PNG"
    igual(fig._suptitle.get_text() if fig._suptitle else "", "Calidad de datos · septiembre 2026")


def prueba_rechazos():
    """Barras de rechazos ordenadas"""
    a = ejes()[2]
    conteo = MOTIVO[MOTIVO != "valido"].value_counts().sort_values()
    igual(sorted(round(b.get_width()) for b in a[0].patches), sorted(conteo.tolist()))
    anchos = [b.get_width() for b in a[0].patches]
    igual(anchos, sorted(anchos), "Ordena de menor a mayor")


def prueba_validos_por_dia():
    """Línea de válidos por día"""
    a = ejes()[2]
    por_dia = VAL.assign(i=F[VAL.index]).set_index("i").resample("D").size()
    igual([int(v) for v in a[1].lines[0].get_ydata()], por_dia.tolist())


def prueba_composicion():
    """Participación por tipo"""
    a = ejes()[2]
    pct = VAL["tipo"].value_counts(normalize=True).reindex(["VOZ", "SMS", "DATOS"]) * 100
    for barra, valor in zip(a[2].patches, pct.tolist()):
        cerca(barra.get_height(), valor, 0.01)


def prueba_gb_por_hora():
    """24 barras de GB por hora"""
    a = ejes()[2]
    datos = VAL[VAL["tipo"] == "DATOS"]
    gb = (datos["mb"].groupby(F[datos.index].dt.hour).sum() / 1024).reindex(range(24), fill_value=0)
    igual(len(a[3].patches), 24)
    cerca(sum(b.get_height() for b in a[3].patches), float(gb.sum()), 0.001)

## Solución

```python
import io

import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

cdr = pd.read_csv("cdr.csv", sep="|", dtype={"linea_origen": "string"}, keep_default_na=False)
clientes = pd.read_csv("clientes.csv", encoding="latin-1", dtype={"linea": "string"})
MOTIVOS = ["duplicado", "linea_vacia", "linea_desconocida", "tipo_desconocido", "duracion_negativa", "fecha_invalida"]


def tablero(cdr, clientes):
    fecha = pd.to_datetime(cdr["fecha_inicio"], format="%Y-%m-%d %H:%M:%S", errors="coerce")
    condiciones = [cdr.duplicated("cdr_id"), cdr["linea_origen"] == "", ~cdr["linea_origen"].isin(clientes["linea"]),
                   ~cdr["tipo"].isin(["VOZ", "SMS", "DATOS"]), cdr["duracion_seg"] < 0, fecha.isna()]
    motivo = pd.Series(np.select(condiciones, MOTIVOS, default="valido"), index=cdr.index)
    validos = cdr[motivo == "valido"].assign(inicio=fecha[motivo == "valido"])

    fig, ejes = plt.subplots(2, 2, figsize=(12, 7), constrained_layout=True)
    fig.suptitle("Calidad de datos · septiembre 2026", fontweight="bold")

    rechazos = motivo[motivo != "valido"].value_counts().sort_values()
    ejes[0, 0].barh(rechazos.index, rechazos.values, color="#dc2626")
    ejes[0, 0].set_title("Registros rechazados por motivo")

    por_dia = validos.set_index("inicio").resample("D").size()
    ejes[0, 1].plot(por_dia.index, por_dia.values, color="#16a34a")
    ejes[0, 1].set_title("Registros válidos por día")
    ejes[0, 1].set_ylim(bottom=0)
    ejes[0, 1].tick_params(axis="x", rotation=45)

    composicion = validos["tipo"].value_counts(normalize=True).reindex(["VOZ", "SMS", "DATOS"], fill_value=0) * 100
    barras = ejes[1, 0].bar(composicion.index, composicion.values, color="#4f46e5")
    ejes[1, 0].bar_label(barras, fmt="%.1f %%")
    ejes[1, 0].set_title("Composición del tráfico válido (%)")

    datos = validos[validos["tipo"] == "DATOS"]
    gb = (datos["mb"].groupby(datos["inicio"].dt.hour).sum() / 1024).reindex(range(24), fill_value=0)
    ejes[1, 1].bar(gb.index, gb.values, color="#f59e0b")
    ejes[1, 1].set_title("GB por hora del día")
    ejes[1, 1].set_xticks(range(0, 24, 3))

    buffer = io.BytesIO()
    fig.savefig(buffer, format="png", dpi=100)
    return fig, buffer.getvalue()


fig, png = tablero(cdr, clientes)
print(len(png), "bytes")
```

## Explicación

Separar **cálculo** (clasificación, agregaciones) de **dibujo** (los cuatro ejes) hace que el tablero sea fácil de mantener: si mañana cambia una regla, el gráfico no se toca.

Devolver los bytes PNG convierte la figura en algo que puedes guardar en disco, adjuntar a un correo o incrustar en un reporte HTML (`base64`). Es exactamente lo que hace esta página para mostrarte los gráficos que genera tu código.

`constrained_layout=True` reparte el espacio entre los cuatro gráficos para que títulos y etiquetas no se monten.
